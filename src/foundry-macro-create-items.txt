// ================================================================
// MACRO FOUNDRY: Crea Oggetti dal sito → Foundry  (Exanthia Web)
// ================================================================
// Specchio inverso della macro di sincronizzazione personaggio.
// Legge la coda `foundry_inbox` su Firestore (gli oggetti creati dal
// form "Crea Oggetto → Foundry" sul sito) e li CREA come Item dnd5e:
//   - target "world"  → nella cartella Oggetti del mondo
//   - target "player" → nell'inventario del personaggio (via firebaseUID)
// Dopo aver creato un oggetto, lo RIMUOVE dalla coda.
//
// SPESE DELL'OFFICINA (2026-09-22): le voci con `kind: "gold"` non sono
// oggetti ma MONETE da togliere (o aggiungere, con op "add") al personaggio:
// sono i materiali del crafting, mandati dal Master dalla sezione
// "Spese dei materiali" nell'Officina. La macro scala le monete d'oro
// dell'attore, scrive un messaggio in chat e cancella la voce dalla coda.
// Guarda anche DENTRO LE BORSE: in dnd5e i contenitori hanno una loro
// `system.currency`, e senza frugarci un PG che tiene l'oro nello zaino
// risultava a zero. Si paga prima con le tasche, poi con le borse.
//
// E' L'UNICA MACRO CHE SERVE per l'Officina: oggetti da aggiungere E monete
// da togliere passano tutti da qui. (La "sync" e' l'altra direzione: porta la
// scheda Foundry sul sito.)
//
// USO: lanciala come GM. Con AUTO_OGNI_MINUTI > 0 resta accesa da sola e
// ricontrolla la coda ogni N minuti finche' il tuo Foundry e' aperto (se
// ricarichi la pagina, rilanciala una volta). Rilanciarla non crea doppioni.
//
// AUTOMATICO (2026-10-03): le spese entrano in coda DA SOLE quando il
// giocatore tira al banco; quando la macro le scala su Foundry scrive anche
// il nuovo oro sul sito (characters/{uid}.currency.gp) e toglie la spesa da
// `crafting.goldPending`. Niente piu' bottoni "Manda a Foundry" per il Master.
//
// NOTA: gli oggetti "player" (e le spese in monete) vanno nell'attore che ha
//   il flag `world.firebaseUID` uguale all'UID del personaggio sul sito.
//   Se esce "non e' collegato a nessun attore", quel flag manca: lancia una
//   volta la macro "Collega attori al sito" (src/foundry-macro-link-actors.txt),
//   che lo scrive su tutti gli attori abbinandoli per nome, e poi rilancia
//   questa. Finche' il flag manca la coda NON viene svuotata: non si perde
//   niente, si riprova e basta.
// ================================================================

// --- CONFIGURAZIONE FIREBASE (uguale all'altra macro) ---
const FIREBASE_CONFIG = {
  projectId: "TUO_PROJECT_ID",
  apiKey:    "TUA_API_KEY"
};

// Da dove si attinge per primo quando si paga: true = prima le tasche del PG
// e poi le borse, false = prima le borse. (Le monete IN PIU' vanno sempre
// addosso al PG, mai dentro una borsa.)
const PRIMA_LE_TASCHE = true;

// Ogni quanti minuti ricontrollare la coda da sola (0 = solo quando la lanci).
const AUTO_OGNI_MINUTI = 2;

const BASE = `https://firestore.googleapis.com/v1/projects/${FIREBASE_CONFIG.projectId}/databases/(default)/documents`;

// ----------------------------------------------------------------
// HELPER: converte un valore Firestore REST in JS semplice
// ----------------------------------------------------------------
function val(v) {
  if (v.stringValue !== undefined) return v.stringValue;
  if (v.integerValue !== undefined) return Number(v.integerValue);
  if (v.doubleValue !== undefined) return Number(v.doubleValue);
  if (v.booleanValue !== undefined) return v.booleanValue;
  if (v.nullValue !== undefined) return null;
  if (v.timestampValue !== undefined) return v.timestampValue;
  if (v.mapValue) return obj(v.mapValue.fields || {});
  if (v.arrayValue) return (v.arrayValue.values || []).map(val);
  return null;
}
function obj(fields) {
  const o = {};
  for (const [k, v] of Object.entries(fields)) o[k] = val(v);
  return o;
}

// ----------------------------------------------------------------
// HELPER: costruisce il `system` dnd5e a partire dai dati del sito
// Gestisce dnd5e v2 e v3 (differenze su weight e properties).
// ----------------------------------------------------------------
const DND_MAJOR = parseInt((game.system.version || "3").split(".")[0]) || 3;

function buildItemData(d) {
  const descHtml = /<[a-z][\s\S]*>/i.test(d.description || "")
    ? (d.description || "")
    : `<p>${(d.description || "").replace(/\n/g, "<br>")}</p>`;

  const system = {
    description: { value: descHtml, chat: "" },
    quantity: d.quantity || 1,
    price: { value: d.price || 0, denomination: "gp" },
    rarity: d.rarity || "",
    identified: true,
  };

  // peso: v3 vuole un oggetto {value, units}, v2 un numero
  system.weight = DND_MAJOR >= 3 ? { value: d.weight || 0, units: "lb" } : (d.weight || 0);

  // ── danno / attacco (armi e consumabili offensivi) ──
  const parts = [];
  if (d.damageFormula) parts.push([d.damageFormula, d.damageType || ""]);
  if (d.damage2Formula) parts.push([d.damage2Formula, d.damage2Type || ""]);
  if (parts.length || d.versatileFormula) {
    system.damage = { parts, versatile: d.versatileFormula || "" };
  }
  if (d.actionType) system.actionType = d.actionType;
  if (d.attackBonus) system.attackBonus = String(d.attackBonus);

  // ── tiro salvezza ──
  if (d.saveAbility && d.saveDC) {
    system.save = { ability: d.saveAbility, dc: d.saveDC, scaling: "flat" };
  }

  // ── per tipo ──
  if (d.foundryType === "weapon") {
    // competenza: v3 usa system.proficient (numero), v2 idem
    system.proficient = d.proficient ? 1 : 0;
    const props = Array.isArray(d.properties) ? d.properties : [];
    if (DND_MAJOR >= 3) {
      system.properties = props;                       // v3: array di chiavi
    } else {
      system.properties = props.reduce((o, k) => (o[k] = true, o), {}); // v2: oggetto
    }
  }

  if (d.foundryType === "equipment" && d.armorType) {
    system.armor = { value: d.armorValue || 0, type: d.armorType, dex: null };
    system.type = { value: d.armorType };
  }

  const out = { name: d.name || "Oggetto", type: d.foundryType || "loot", system };
  if (d.img) out.img = d.img;
  return out;
}

// ----------------------------------------------------------------
// TROVA L'ATTORE DI UN PERSONAGGIO DEL SITO (2026-09-24)
// 1) flag world.firebaseUID sull'attore (la via normale);
// 2) se manca, chiede a Firestore il personaggio characters/{uid}, che la
//    macro di sincronizzazione ha segnato con foundryActorId/foundryUser:
//    a) l'attore con quell'id;  b) l'unico attore "character" di cui
//    l'utente foundryUser e' proprietario;  c) l'unico attore con lo
//    stesso nome del personaggio.
// Quando lo trova per le vie 2, SCRIVE il flag: dalla volta dopo e' la via 1.
// ----------------------------------------------------------------
const _normNome = (x) => String(x ?? "").normalize("NFD").replace(/[̀-ͯ]/g, "")
  .toLowerCase().replace(/[´`'’ʼ]/g, "'").replace(/\s+/g, " ").trim();
const _cacheAttori = new Map();

async function trovaAttore(uid) {
  if (!uid) return null;
  if (_cacheAttori.has(uid)) return _cacheAttori.get(uid);
  const pg = game.actors.contents.filter((a) => a.type === "character");

  let actor = game.actors.contents.find((a) => a.getFlag("world", "firebaseUID") === uid) || null;
  let via = "flag";

  if (!actor) {
    let c = {};
    try {
      const res = await fetch(`${BASE}/characters/${uid}?key=${FIREBASE_CONFIG.apiKey}`
        + "&mask.fieldPaths=name&mask.fieldPaths=foundryActorId&mask.fieldPaths=foundryUser");
      if (res.ok) c = obj((await res.json()).fields || {});
    } catch (e) { console.warn("[Import] Lettura personaggio fallita:", e); }

    // a) id dell'attore scritto dalla sync (scartato se e' gia' di un altro)
    const perId = c.foundryActorId ? game.actors.get(c.foundryActorId) : null;
    if (perId && !perId.getFlag("world", "firebaseUID")) { actor = perId; via = "id della sync"; }

    // b) l'utente di Foundry proprietario
    if (!actor && c.foundryUser) {
      const u = game.users.find((x) => x.name === c.foundryUser);
      const suoi = u ? pg.filter((a) => (a.ownership?.[u.id] ?? 0) >= 3
        && !a.getFlag("world", "firebaseUID")) : [];
      // piu' attori dello stesso utente: si sceglie quello col nome del PG
      const scelti = suoi.length > 1 ? suoi.filter((a) => _normNome(a.name) === _normNome(c.name)
        || _normNome(a.name).split(" ")[0] === _normNome(c.name).split(" ")[0]) : suoi;
      if (scelti.length === 1) { actor = scelti[0]; via = `proprietario ${u.name}`; }
      else if (suoi.length > 1) console.warn(`[Import] ${c.foundryUser} possiede ${suoi.length} attori: ${suoi.map((a) => a.name).join(", ")}. Sincronizza la scheda giusta per sceglierla.`);
    }

    // c) stesso nome (intero, poi solo il primo nome)
    if (!actor && c.name) {
      const liberi = pg.filter((a) => !a.getFlag("world", "firebaseUID"));
      let omonimi = liberi.filter((a) => _normNome(a.name) === _normNome(c.name));
      if (!omonimi.length) omonimi = liberi.filter((a) => _normNome(a.name).split(" ")[0] === _normNome(c.name).split(" ")[0]);
      if (omonimi.length === 1) { actor = omonimi[0]; via = "nome"; }
    }
    if (!actor) console.warn(`[Import] Attore non trovato per "${c.name || uid}" (uid ${uid}). Utente Foundry sul sito: ${c.foundryUser || "nessuno"}. Lancia la sync dalla scheda di quel PG, oppure mettigli il flag a mano: game.actors.getName("NOME").setFlag("world","firebaseUID","${uid}")`);

    if (actor) {
      await actor.setFlag("world", "firebaseUID", uid);
      console.log(`[Import] ${actor.name} collegato a ${c.name || uid} (via ${via}), per sempre.`);
      ui.notifications.info(`🔗 ${actor.name} collegato al sito (${via}).`);
    }
  }

  _cacheAttori.set(uid, actor);
  return actor;
}

// ----------------------------------------------------------------
// SITO: dopo aver scalato le monete su Foundry, scrive sul personaggio del
// sito l'oro nuovo (totale tasche + borse, come fa la sync) e, per le spese
// automatiche dell'Officina, toglie l'importo da crafting.goldPending.
// Una sola scrittura atomica (commit con increment).
// ----------------------------------------------------------------
async function aggiornaSito(uid, gpDopo, scalaPending) {
  const nome = `projects/${FIREBASE_CONFIG.projectId}/databases/(default)/documents/characters/${uid}`;
  const write = {
    update: { name: nome, fields: { currency: { mapValue: { fields: { gp: { integerValue: String(Math.max(0, Math.round(gpDopo))) } } } } } },
    updateMask: { fieldPaths: ["currency.gp"] },
    currentDocument: { exists: true },
  };
  if (scalaPending > 0) {
    write.updateTransforms = [{ fieldPath: "crafting.goldPending", increment: { integerValue: String(-Math.round(scalaPending)) } }];
  }
  const res = await fetch(`https://firestore.googleapis.com/v1/projects/${FIREBASE_CONFIG.projectId}/databases/(default)/documents:commit?key=${FIREBASE_CONFIG.apiKey}`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ writes: [write] }),
  });
  if (!res.ok) throw new Error(`sito non aggiornato (${res.status}): ${await res.text()}`);
}

// ----------------------------------------------------------------
// MAIN
// ----------------------------------------------------------------
async function importaDalSito(silenziosa = false) {
  if (!silenziosa) ui.notifications.info("Importazione oggetti dal sito in corso…");

  // 1) leggi la coda
  let docs = [];
  try {
    const res = await fetch(`${BASE}/foundry_inbox?key=${FIREBASE_CONFIG.apiKey}&pageSize=200`);
    const data = await res.json();
    docs = (data.documents || []).map(doc => ({
      _path: doc.name,                         // path completo per DELETE
      _id: doc.name.split("/").pop(),
      ...obj(doc.fields || {})
    }));
  } catch (e) {
    if (!silenziosa) ui.notifications.error("Errore lettura coda Firestore.");
    console.error(e);
    return;
  }

  const pending = docs.filter(d => d.status === "pending");
  if (!pending.length) {
    if (!silenziosa) ui.notifications.info("Nessun oggetto in coda.");
    return;
  }

  let created = 0, skipped = 0, gold = 0;
  for (const d of pending) {
    try {
      // ── SPESA DI MONETE (Officina): non è un oggetto ──
      if (d.kind === "gold") {
        const actor = await trovaAttore(d.targetUid);
        if (!actor) {
          console.warn(`[Import] Nessun attore con firebaseUID ${d.targetUid} per la spesa di ${d.amount} mo. Salto.`);
          ui.notifications.warn(`Spesa di ${d.amount} mo: ${d.targetName || d.targetUid} non e' collegato a nessun attore. Sincronizza una volta la sua scheda (macro di sync) e rilancia.`);
          skipped++;
          continue; // lascia in coda per riprovare
        }
        const amount = Math.max(0, Number(d.amount) || 0);
        const add = d.op === "add";

        // ── Le CASSE da cui prendere: le tasche del PG e poi ogni borsa.
        // In dnd5e i contenitori (zaini, borse, sacchetti) hanno una loro
        // `system.currency`: senza guardarci dentro un PG con l'oro nello
        // zaino risultava al verde e la spesa finiva clampata a 0.
        const borse = actor.items.filter((i) =>
          (i.type === "container" || i.type === "backpack") &&
          foundry.utils.getProperty(i, "system.currency"));
        const casse = [
          { nome: "tasche", leggi: () => Number(foundry.utils.getProperty(actor, "system.currency.gp")) || 0,
            scrivi: (v) => actor.update({ "system.currency.gp": v }) },
          ...borse.map((b) => ({ nome: b.name,
            leggi: () => Number(foundry.utils.getProperty(b, "system.currency.gp")) || 0,
            scrivi: (v) => b.update({ "system.currency.gp": v }) })),
        ];
        // L'ordine: si paga con quel che si ha addosso, poi si frugano le
        // borse. Metti PRIMA_LE_TASCHE a false per invertirlo.
        if (!PRIMA_LE_TASCHE) casse.reverse();

        const prima = casse.reduce((a, c) => a + c.leggi(), 0);
        const mosse = [];
        let resto = amount;

        if (add) {
          // Le monete in piu' vanno sempre addosso al PG, non in una borsa.
          const t = casse.find((c) => c.nome === "tasche") || casse[0];
          await t.scrivi(t.leggi() + amount);
          mosse.push(`${t.nome} +${amount}`);
          resto = 0;
        } else {
          for (const c of casse) {
            if (resto <= 0) break;
            const ha = c.leggi();
            if (ha <= 0) continue;
            const preso = Math.min(ha, resto);
            await c.scrivi(ha - preso);
            resto -= preso;
            mosse.push(`${c.nome} −${preso}`);
          }
        }

        const dopo = casse.reduce((a, c) => a + c.leggi(), 0);
        const mancano = resto; // > 0 solo se non bastavano nemmeno le borse
        await ChatMessage.create({
          speaker: { alias: "Officina" },
          content: `<p><strong>${actor.name}</strong>: ${add ? "+" : "−"}${amount} mo (${prima} → ${dopo}).</p>`
            + (mosse.length > 1 ? `<p style="font-size:.9em;opacity:.8">Presi da: ${mosse.join(", ")}.</p>` : "")
            + `<p style="font-size:.9em;opacity:.8">${d.reason || ""}</p>`
            + (mancano ? `<p style="color:#b91c1c"><strong>Attenzione:</strong> gli mancavano ${mancano} mo, tasche e borse sono a 0.</p>` : ""),
          whisper: ChatMessage.getWhisperRecipients("GM").map(u => u.id),
        });
        console.log(`[Import] ${add ? "+" : "−"}${amount} mo → ${actor.name} (${prima} → ${dopo})`, mosse);
        // Il sito si allinea subito (senza aspettare la sync della scheda).
        // Se non ci riesce la voce si cancella lo stesso: l'oro su Foundry e'
        // gia' stato scalato e rifarlo sarebbe peggio. Lo sistema la sync.
        try {
          await aggiornaSito(d.targetUid, dopo, d.pendingOnSite && !add ? amount : 0);
        } catch (e) {
          console.warn("[Import] Oro scalato su Foundry ma il sito non si e' aggiornato:", e);
          ui.notifications.warn(`${actor.name}: oro scalato su Foundry, ma il sito non si e' aggiornato. Lancia la sync della sua scheda.`);
        }
        await fetch(`https://firestore.googleapis.com/v1/${d._path}?key=${FIREBASE_CONFIG.apiKey}`, { method: "DELETE" });
        gold++;
        continue;
      }

      const itemData = buildItemData(d);

      if (d.target === "player") {
        // trova l'attore col flag firebaseUID corrispondente
        const actor = await trovaAttore(d.targetUid);
        if (!actor) {
          console.warn(`[Import] Nessun attore con firebaseUID ${d.targetUid} per "${d.name}". Salto.`);
          ui.notifications.warn(`"${d.name}": ${d.targetName || d.targetUid} non e' collegato a nessun attore. Sincronizza una volta la sua scheda (macro di sync) e rilancia.`);
          skipped++;
          continue; // lascia in coda per riprovare
        }
        await actor.createEmbeddedDocuments("Item", [itemData]);
        console.log(`[Import] "${d.name}" → inventario di ${actor.name}`);
      } else {
        await Item.create(itemData);
        console.log(`[Import] "${d.name}" → cartella Oggetti del mondo`);
      }

      // 2) rimuovi dalla coda (DELETE su Firestore)
      await fetch(`https://firestore.googleapis.com/v1/${d._path}?key=${FIREBASE_CONFIG.apiKey}`, { method: "DELETE" });
      created++;
    } catch (e) {
      console.error(`[Import] Errore su "${d.name}":`, e);
      ui.notifications.error(`Errore su "${d.name}": ${e.message || e}`);
      skipped++;
    }
  }

  if (!silenziosa || created || gold || skipped) {
    ui.notifications.info(`✅ Creati ${created} oggetti${gold ? `, ${gold} spese scalate` : ""}${skipped ? `, ${skipped} saltati` : ""}.`);
  }
}

// ----------------------------------------------------------------
// AVVIO: un giro subito e, se richiesto, uno ogni AUTO_OGNI_MINUTI.
// Il timer vecchio si spegne prima di accenderne uno nuovo, e un giro non
// parte se il precedente non e' ancora finito: niente doppioni.
// ----------------------------------------------------------------
(async () => {
  if (!game.user.isGM) { ui.notifications.warn("Questa macro la lancia solo il GM."); return; }
  const G = (window.__eldoriaImport ||= { timer: null, inCorso: false });
  const giro = async (silenziosa) => {
    if (G.inCorso) return;
    G.inCorso = true;
    try { await importaDalSito(silenziosa); }
    catch (e) { console.error("[Import] giro fallito:", e); }
    finally { G.inCorso = false; }
  };
  if (G.timer) { clearInterval(G.timer); G.timer = null; }
  await giro(false);
  if (AUTO_OGNI_MINUTI > 0) {
    G.timer = setInterval(() => giro(true), AUTO_OGNI_MINUTI * 60 * 1000);
    ui.notifications.info(`🔁 Importazione automatica accesa: ogni ${AUTO_OGNI_MINUTI} minuti finche' Foundry resta aperto.`);
  }
})();
