// ================================================================
// MACRO FOUNDRY: Crea Oggetti dal sito → Foundry  (Exanthia Web)
// ================================================================
// Specchio inverso della macro di sincronizzazione personaggio.
// Legge la coda `foundry_inbox` su Firestore (gli oggetti creati dal
// form "Crea Oggetto → Foundry" sul sito) e li CREA come Item dnd5e:
//   - target "world"  → nella cartella Oggetti del mondo
//   - target "player" → nell'inventario del personaggio (via firebaseUID)
// Dopo aver creato un oggetto, lo RIMUOVE dalla coda.
//
// SPESE DELL'OFFICINA (2026-09-22): le voci con `kind: "gold"` non sono
// oggetti ma MONETE da togliere (o aggiungere, con op "add") al personaggio:
// sono i materiali del crafting, mandati dal Master dalla sezione
// "Spese dei materiali" nell'Officina. La macro scala le monete d'oro
// dell'attore, scrive un messaggio in chat e cancella la voce dalla coda.
// Guarda anche DENTRO LE BORSE: in dnd5e i contenitori hanno una loro
// `system.currency`, e senza frugarci un PG che tiene l'oro nello zaino
// risultava a zero. Si paga prima con le tasche, poi con le borse.
//
// USO: lanciala come GM quando vuoi importare gli oggetti in coda.
// (Funziona identica su Foundry online: la macro gira nel tuo browser.)
//
// NOTA: gli oggetti "player" (e le spese in monete) vanno nell'attore che ha
//   il flag `world.firebaseUID` uguale all'UID del personaggio sul sito.
//   Se esce "non e' collegato a nessun attore", quel flag manca: lancia una
//   volta la macro "Collega attori al sito" (src/foundry-macro-link-actors.txt),
//   che lo scrive su tutti gli attori abbinandoli per nome, e poi rilancia
//   questa. Finche' il flag manca la coda NON viene svuotata: non si perde
//   niente, si riprova e basta.
// ================================================================

// --- CONFIGURAZIONE FIREBASE (uguale all'altra macro) ---
const FIREBASE_CONFIG = {
  projectId: "TUO_PROJECT_ID",
  apiKey:    "TUA_API_KEY"
};

// Da dove si attinge per primo quando si paga: true = prima le tasche del PG
// e poi le borse, false = prima le borse. (Le monete IN PIU' vanno sempre
// addosso al PG, mai dentro una borsa.)
const PRIMA_LE_TASCHE = true;

const BASE = `https://firestore.googleapis.com/v1/projects/${FIREBASE_CONFIG.projectId}/databases/(default)/documents`;

// ----------------------------------------------------------------
// HELPER: converte un valore Firestore REST in JS semplice
// ----------------------------------------------------------------
function val(v) {
  if (v.stringValue !== undefined) return v.stringValue;
  if (v.integerValue !== undefined) return Number(v.integerValue);
  if (v.doubleValue !== undefined) return Number(v.doubleValue);
  if (v.booleanValue !== undefined) return v.booleanValue;
  if (v.nullValue !== undefined) return null;
  if (v.timestampValue !== undefined) return v.timestampValue;
  if (v.mapValue) return obj(v.mapValue.fields || {});
  if (v.arrayValue) return (v.arrayValue.values || []).map(val);
  return null;
}
function obj(fields) {
  const o = {};
  for (const [k, v] of Object.entries(fields)) o[k] = val(v);
  return o;
}

// ----------------------------------------------------------------
// HELPER: costruisce il `system` dnd5e a partire dai dati del sito
// Gestisce dnd5e v2 e v3 (differenze su weight e properties).
// ----------------------------------------------------------------
const DND_MAJOR = parseInt((game.system.version || "3").split(".")[0]) || 3;

function buildItemData(d) {
  const descHtml = /<[a-z][\s\S]*>/i.test(d.description || "")
    ? (d.description || "")
    : `<p>${(d.description || "").replace(/\n/g, "<br>")}</p>`;

  const system = {
    description: { value: descHtml, chat: "" },
    quantity: d.quantity || 1,
    price: { value: d.price || 0, denomination: "gp" },
    rarity: d.rarity || "",
    identified: true,
  };

  // peso: v3 vuole un oggetto {value, units}, v2 un numero
  system.weight = DND_MAJOR >= 3 ? { value: d.weight || 0, units: "lb" } : (d.weight || 0);

  // ── danno / attacco (armi e consumabili offensivi) ──
  const parts = [];
  if (d.damageFormula) parts.push([d.damageFormula, d.damageType || ""]);
  if (d.damage2Formula) parts.push([d.damage2Formula, d.damage2Type || ""]);
  if (parts.length || d.versatileFormula) {
    system.damage = { parts, versatile: d.versatileFormula || "" };
  }
  if (d.actionType) system.actionType = d.actionType;
  if (d.attackBonus) system.attackBonus = String(d.attackBonus);

  // ── tiro salvezza ──
  if (d.saveAbility && d.saveDC) {
    system.save = { ability: d.saveAbility, dc: d.saveDC, scaling: "flat" };
  }

  // ── per tipo ──
  if (d.foundryType === "weapon") {
    // competenza: v3 usa system.proficient (numero), v2 idem
    system.proficient = d.proficient ? 1 : 0;
    const props = Array.isArray(d.properties) ? d.properties : [];
    if (DND_MAJOR >= 3) {
      system.properties = props;                       // v3: array di chiavi
    } else {
      system.properties = props.reduce((o, k) => (o[k] = true, o), {}); // v2: oggetto
    }
  }

  if (d.foundryType === "equipment" && d.armorType) {
    system.armor = { value: d.armorValue || 0, type: d.armorType, dex: null };
    system.type = { value: d.armorType };
  }

  const out = { name: d.name || "Oggetto", type: d.foundryType || "loot", system };
  if (d.img) out.img = d.img;
  return out;
}

// ----------------------------------------------------------------
// MAIN
// ----------------------------------------------------------------
(async () => {
  ui.notifications.info("Importazione oggetti dal sito in corso…");

  // 1) leggi la coda
  let docs = [];
  try {
    const res = await fetch(`${BASE}/foundry_inbox?key=${FIREBASE_CONFIG.apiKey}&pageSize=200`);
    const data = await res.json();
    docs = (data.documents || []).map(doc => ({
      _path: doc.name,                         // path completo per DELETE
      _id: doc.name.split("/").pop(),
      ...obj(doc.fields || {})
    }));
  } catch (e) {
    ui.notifications.error("Errore lettura coda Firestore.");
    console.error(e);
    return;
  }

  const pending = docs.filter(d => d.status === "pending");
  if (!pending.length) {
    ui.notifications.info("Nessun oggetto in coda.");
    return;
  }

  let created = 0, skipped = 0, gold = 0;
  for (const d of pending) {
    try {
      // ── SPESA DI MONETE (Officina): non è un oggetto ──
      if (d.kind === "gold") {
        const actor = game.actors.contents.find(a => a.getFlag("world", "firebaseUID") === d.targetUid);
        if (!actor) {
          console.warn(`[Import] Nessun attore con firebaseUID ${d.targetUid} per la spesa di ${d.amount} mo. Salto.`);
          ui.notifications.warn(`Spesa di ${d.amount} mo: ${d.targetName || d.targetUid} non e' collegato a nessun attore. Lancia la macro "Collega attori al sito".`);
          skipped++;
          continue; // lascia in coda per riprovare
        }
        const amount = Math.max(0, Number(d.amount) || 0);
        const add = d.op === "add";

        // ── Le CASSE da cui prendere: le tasche del PG e poi ogni borsa.
        // In dnd5e i contenitori (zaini, borse, sacchetti) hanno una loro
        // `system.currency`: senza guardarci dentro un PG con l'oro nello
        // zaino risultava al verde e la spesa finiva clampata a 0.
        const borse = actor.items.filter((i) =>
          (i.type === "container" || i.type === "backpack") &&
          foundry.utils.getProperty(i, "system.currency"));
        const casse = [
          { nome: "tasche", leggi: () => Number(foundry.utils.getProperty(actor, "system.currency.gp")) || 0,
            scrivi: (v) => actor.update({ "system.currency.gp": v }) },
          ...borse.map((b) => ({ nome: b.name,
            leggi: () => Number(foundry.utils.getProperty(b, "system.currency.gp")) || 0,
            scrivi: (v) => b.update({ "system.currency.gp": v }) })),
        ];
        // L'ordine: si paga con quel che si ha addosso, poi si frugano le
        // borse. Metti PRIMA_LE_TASCHE a false per invertirlo.
        if (!PRIMA_LE_TASCHE) casse.reverse();

        const prima = casse.reduce((a, c) => a + c.leggi(), 0);
        const mosse = [];
        let resto = amount;

        if (add) {
          // Le monete in piu' vanno sempre addosso al PG, non in una borsa.
          const t = casse.find((c) => c.nome === "tasche") || casse[0];
          await t.scrivi(t.leggi() + amount);
          mosse.push(`${t.nome} +${amount}`);
          resto = 0;
        } else {
          for (const c of casse) {
            if (resto <= 0) break;
            const ha = c.leggi();
            if (ha <= 0) continue;
            const preso = Math.min(ha, resto);
            await c.scrivi(ha - preso);
            resto -= preso;
            mosse.push(`${c.nome} −${preso}`);
          }
        }

        const dopo = casse.reduce((a, c) => a + c.leggi(), 0);
        const mancano = resto; // > 0 solo se non bastavano nemmeno le borse
        await ChatMessage.create({
          speaker: { alias: "Officina" },
          content: `<p><strong>${actor.name}</strong>: ${add ? "+" : "−"}${amount} mo (${prima} → ${dopo}).</p>`
            + (mosse.length > 1 ? `<p style="font-size:.9em;opacity:.8">Presi da: ${mosse.join(", ")}.</p>` : "")
            + `<p style="font-size:.9em;opacity:.8">${d.reason || ""}</p>`
            + (mancano ? `<p style="color:#b91c1c"><strong>Attenzione:</strong> gli mancavano ${mancano} mo, tasche e borse sono a 0.</p>` : ""),
          whisper: ChatMessage.getWhisperRecipients("GM").map(u => u.id),
        });
        console.log(`[Import] ${add ? "+" : "−"}${amount} mo → ${actor.name} (${prima} → ${dopo})`, mosse);
        await fetch(`https://firestore.googleapis.com/v1/${d._path}?key=${FIREBASE_CONFIG.apiKey}`, { method: "DELETE" });
        gold++;
        continue;
      }

      const itemData = buildItemData(d);

      if (d.target === "player") {
        // trova l'attore col flag firebaseUID corrispondente
        const actor = game.actors.contents.find(a => a.getFlag("world", "firebaseUID") === d.targetUid);
        if (!actor) {
          console.warn(`[Import] Nessun attore con firebaseUID ${d.targetUid} per "${d.name}". Salto.`);
          ui.notifications.warn(`"${d.name}": ${d.targetName || d.targetUid} non e' collegato a nessun attore. Lancia la macro "Collega attori al sito".`);
          skipped++;
          continue; // lascia in coda per riprovare
        }
        await actor.createEmbeddedDocuments("Item", [itemData]);
        console.log(`[Import] "${d.name}" → inventario di ${actor.name}`);
      } else {
        await Item.create(itemData);
        console.log(`[Import] "${d.name}" → cartella Oggetti del mondo`);
      }

      // 2) rimuovi dalla coda (DELETE su Firestore)
      await fetch(`https://firestore.googleapis.com/v1/${d._path}?key=${FIREBASE_CONFIG.apiKey}`, { method: "DELETE" });
      created++;
    } catch (e) {
      console.error(`[Import] Errore su "${d.name}":`, e);
      ui.notifications.error(`Errore su "${d.name}": ${e.message || e}`);
      skipped++;
    }
  }

  ui.notifications.info(`✅ Creati ${created} oggetti${gold ? `, ${gold} spese scalate` : ""}${skipped ? `, ${skipped} saltati` : ""}.`);
})();
