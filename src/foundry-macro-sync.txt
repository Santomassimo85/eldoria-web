// ================================================================
// MACRO FOUNDRY: Sincronizza Scheda → Crit Happens (versione DM, 2026-09-20)
// ================================================================
// Copia di riferimento della macro che il Master tiene su Foundry: qui
// SENZA la chiave API e SENZA la mappa giocatore→UID (repo pubblico).
// Esporta in `characters/{uid}`: name, class, subclass, race, background,
// image, level, stats (hp/ac/prof/… + str/dex/con/int/wis/cha come
// {score, mod, save}), currency, spellSlots, skills, actions
// (armi/incantesimi/talenti con category/bonus/damage/description) e
// `inventory` (weapon/equipment/consumable/tool/loot/container/backpack
// con name/type/quantity/weight/price/equipped/container/img/description).
// L'Officina del crafting (src/data/craftingOwnership.js) legge `inventory`
// e `actions` per capire se il PG possiede strumenti e componenti.
// ================================================================

/* =========================================================================
   SYNC SCHEDA -> FIRESTORE  (versione DM)
   - La usa il MASTER: seleziona il token, sceglie il giocatore da una tendina.
   - Legge la scheda COMPLETA: statistiche, caratteristiche (punteggio+mod+TS),
     oro (TASCHE + BORSE: i contenitori dnd5e hanno una loro currency),
     slot incantesimo (inclusi i Pact Slot), abilita/skill,
     azioni (armi/incantesimi/feature) e inventario/contenitori.
   ========================================================================= */

const FIREBASE_CONFIG = {
  apiKey: "TUA_API_KEY",
  projectId: "eldoria-web"
};

// Nome giocatore (come appare in Foundry) -> UID Firebase.
// Per aggiungere un giocatore: copia il suo UID da Firebase > Authentication.
const uidMap = {
  // "NomeUtenteFoundry": "UID_FIREBASE",   ← compila in Foundry, non nel repo
};

/* ---- Helper: valori nel formato tipizzato di Firestore ---- */
const fsStr  = v => ({ stringValue: String(v ?? "") });
const fsInt  = v => ({ integerValue: String(Math.trunc(Number(v) || 0)) });
const fsNum  = v => ({ doubleValue: Number(v) || 0 });
const fsBool = v => ({ booleanValue: !!v });
const fsMap  = fields => ({ mapValue: { fields } });
const fsArr  = values => ({ arrayValue: { values } });

/* ---- Helper: pulisce le descrizioni dalla "roba di Foundry" ---- */
function cleanText(html) {
  if (!html) return "Nessuna descrizione.";
  let t = String(html);
  t = t.replace(/<[^>]*>/g, " ");                                  // tag HTML
  t = t.replace(/&nbsp;/gi, " ").replace(/&amp;/gi, "&");          // entita'
  t = t.replace(/@\w+\[([^\]|]+)(?:\|[^\]]*)?\]/g, "$1");          // @spell[x|y] -> x
  t = t.replace(/&Reference\[(?:[^\]=]*=)?([^\]{}]+)\]/gi, "$1");  // &Reference[...] -> parola
  t = t.replace(/\{([^}]*)\}/g, "$1");                            // {etichetta} -> etichetta
  t = t.replace(/\[\[[^\]]*\]\]/g, "");                          // [[/damage ...]]
  t = t.replace(/\s+/g, " ").trim();
  return t || "Nessuna descrizione.";
}

/* ---- Attore selezionato ---- */
const actor = canvas.tokens.controlled[0]?.actor || game.user.character;

if (!actor) {
  ui.notifications.error("Seleziona sulla mappa il token del personaggio da sincronizzare.");
} else {

  const A = actor.system;
  const strMod = A.abilities.str.mod;
  const dexMod = A.abilities.dex.mod;

  /* ---- Finesse / Ranged (invariato) ---- */
  function isFinesse(item) {
    const props = item.system?.properties;
    if (!props) return false;
    if (props instanceof Set) return props.has("fin");
    if (Array.isArray(props)) return props.includes("fin");
    if (typeof props === "object") return props?.fin === true;
    return false;
  }
  function isRanged(item) {
    const actionType = item.system?.actionType;
    const props = item.system?.properties;
    if (actionType === "rwak") return true;
    if (props instanceof Set) return props.has("amm");
    if (typeof props === "object") return props?.amm === true;
    return false;
  }
  const finesseWeapons = actor.items.filter(i => i.type === "weapon" && isFinesse(i));

  async function askFinesseChoices() {
    const choices = {};
    for (const weapon of finesseWeapons) {
      await new Promise((resolve) => {
        new Dialog({
          title: `Arma Accurata: ${weapon.name}`,
          content: `
            <div style="padding:10px">
              <p>Quale caratteristica per <strong>${weapon.name}</strong>?</p>
              <p>💪 Forza: <strong>${strMod >= 0 ? "+" : ""}${strMod}</strong>
                 &nbsp;|&nbsp; 🏃 Destrezza: <strong>${dexMod >= 0 ? "+" : ""}${dexMod}</strong></p>
            </div>`,
          buttons: {
            str: { label: `Forza (${strMod >= 0 ? "+" : ""}${strMod})`,
                   callback: () => { choices[weapon.id] = "str"; resolve(); } },
            dex: { label: `Destrezza (${dexMod >= 0 ? "+" : ""}${dexMod})`,
                   callback: () => { choices[weapon.id] = "dex"; resolve(); } }
          },
          default: strMod >= dexMod ? "str" : "dex",
          close: () => { choices[weapon.id] = strMod >= dexMod ? "str" : "dex"; resolve(); }
        }).render(true);
      });
    }
    return choices;
  }

  function getDamageFormula(item, finesseChoices) {
    if (item.type === "spell") {
      const damages = item.labels?.damages;
      if (damages && damages.length > 0) {
        const formula = damages[0]?.formula;
        if (formula && formula !== "0") return formula;
      }
      return "0";
    }
    if (item.type === "weapon") {
      const baseFormula = item.labels?.damage;
      if (!baseFormula || baseFormula === "0") return "0";
      let mod = 0;
      if (isFinesse(item)) {
        const chosen = finesseChoices[item.id];
        mod = chosen === "dex" ? dexMod : strMod;
      } else if (isRanged(item)) {
        mod = dexMod;
      } else {
        mod = strMod;
      }
      if (mod !== 0) return mod > 0 ? `${baseFormula}+${mod}` : `${baseFormula}${mod}`;
      return baseFormula;
    }
    const generic = item.labels?.damage || item.labels?.damages?.[0]?.formula;
    if (generic && generic !== "0") return generic;
    return "0";
  }

  /* ---- Pre-seleziona il proprietario del token ---- */
  function detectOwnerName() {
    const own = actor.ownership || {};
    for (const [userId, lvl] of Object.entries(own)) {
      if (lvl === 3 && userId !== "default") {
        const u = game.users?.get(userId);
        if (u && uidMap[u.name]) return u.name;
      }
    }
    return null;
  }

  /* ---- Tendina: il DM sceglie il giocatore ---- */
  async function chooseTarget() {
    const def = detectOwnerName();
    const opts = Object.keys(uidMap)
      .sort((a, b) => a.localeCompare(b))
      .map(n => `<option value="${n}" ${n === def ? "selected" : ""}>${n}</option>`)
      .join("");
    return new Promise(resolve => {
      new Dialog({
        title: "Sincronizza scheda (DM)",
        content: `
          <div style="padding:10px">
            <p>Token: <strong>${actor.name}</strong></p>
            <p>A quale giocatore appartiene questa scheda?</p>
            <select id="tgt" style="width:100%;padding:6px;margin-top:4px">${opts}</select>
          </div>`,
        buttons: {
          ok: { icon: '<i class="fas fa-cloud-upload-alt"></i>', label: "Sincronizza",
                callback: html => resolve(html.find("#tgt").val()) },
          no: { icon: '<i class="fas fa-times"></i>', label: "Annulla",
                callback: () => resolve(null) }
        },
        default: "ok",
        close: () => resolve(null)
      }).render(true);
    });
  }

  /* ---- Costruisce i dati e li invia ---- */
  async function run() {
    // uidMap vuota = e' stata incollata la copia del repo (sanificata):
    // la tendina uscirebbe vuota. Serve la versione del Master con la mappa.
    if (!Object.keys(uidMap).length) {
      ui.notifications.error("uidMap e' vuota: questa e' la copia del repo senza giocatori. Incolla la versione DM (foundry-macro-sync-DM.txt) o compila uidMap.", { permanent: true });
      return;
    }
    const targetName = await chooseTarget();
    if (!targetName) { ui.notifications.warn("Sincronizzazione annullata."); return; }
    const firebaseUID = uidMap[targetName];
    if (!firebaseUID) { ui.notifications.error(`Giocatore ${targetName} non mappato.`); return; }

    const finesseChoices = await askFinesseChoices();

    /* AZIONI: armi / incantesimi / feature */
    const actions = actor.items
      .filter(i => ["weapon", "spell", "feat"].includes(i.type))
      .map(i => {
        const damage = getDamageFormula(i, finesseChoices);
        const bonus = i.labels?.toHit ? i.labels.toHit.replace(/[^0-9+-]/g, "") : "0";
        let category = "Abilità";
        if (i.type === "weapon") category = "Armi";
        if (i.type === "spell") {
          const lvl = i.system.level;
          category = lvl === 0 ? "Trucchetto" : `Livello ${lvl}`;
        }
        return fsMap({
          name: fsStr(i.name),
          category: fsStr(category),
          bonus: fsStr(bonus),
          damage: fsStr(damage),
          description: fsStr(cleanText(i.system.description?.value))
        });
      });

    /* INVENTARIO + CONTENITORI (zaini, borse, oggetti) */
    const PHYS = ["weapon", "equipment", "consumable", "tool", "loot", "container", "backpack"];
    const inventory = actor.items
      .filter(i => PHYS.includes(i.type))
      .map(i => {
        const w = typeof i.system.weight === "object" ? i.system.weight?.value : i.system.weight;
        const p = typeof i.system.price === "object" ? i.system.price?.value : i.system.price;
        return fsMap({
          name: fsStr(i.name),
          type: fsStr(i.type),
          quantity: fsInt(i.system.quantity ?? 1),
          weight: fsNum(w ?? 0),
          price: fsNum(p ?? 0),
          equipped: fsBool(i.system.equipped ?? false),
          isContainer: fsBool(i.type === "container" || i.type === "backpack"),
          container: fsStr(i.system.container ?? ""),
          img: fsStr(i.img ?? ""),
          description: fsStr(cleanText(i.system.description?.value))
        });
      });

    /* CARATTERISTICHE: punteggio + modificatore + tiro salvezza */
    const ab = A.abilities || {};
    const abilityMap = {};
    for (const k of ["str", "dex", "con", "int", "wis", "cha"]) {
      const a = ab[k] || {};
      const save = typeof a.save === "object" ? (a.save?.value ?? a.mod) : (a.save ?? a.mod);
      abilityMap[k] = fsMap({
        score: fsInt(a.value ?? 10),
        mod: fsInt(a.mod ?? 0),
        save: fsInt(save ?? 0)
      });
    }

    /* ABILITÀ / SKILL */
    const skills = Object.entries(A.skills || {}).map(([key, sk]) => {
      const raw = CONFIG.DND5E?.skills?.[key]?.label || key.toUpperCase();
      const label = game.i18n?.localize ? game.i18n.localize(raw) : raw;
      return fsMap({
        name: fsStr(label),
        mod: fsInt(sk.total ?? sk.mod ?? 0),
        proficient: fsBool(sk.proficient)
      });
    });

    /* SLOT INCANTESIMO (inclusi i Pact Slot del Warlock) */
    const sp = A.spells || {};
    const slots = [];
    for (let n = 1; n <= 9; n++) {
      const s = sp[`spell${n}`];
      if (s && (s.max ?? 0) > 0) {
        slots.push(fsMap({ label: fsStr(`Livello ${n}`), value: fsInt(s.value ?? 0), max: fsInt(s.max ?? 0) }));
      }
    }
    if (sp.pact && (sp.pact.max ?? 0) > 0) {
      slots.push(fsMap({ label: fsStr(`Patto (liv. ${sp.pact.level ?? "?"})`), value: fsInt(sp.pact.value ?? 0), max: fsInt(sp.pact.max ?? 0) }));
    }

    /* ORO / MONETE: TASCHE + BORSE (2026-09-22)
       In dnd5e i contenitori (zaini, borse, sacchetti) hanno una LORO
       `system.currency`: quelle monete non stanno in `actor.system.currency`.
       Il sito deve vedere il TOTALE, altrimenti un PG che tiene l'oro nello
       zaino risulta al verde e l'Officina non lo fa lavorare.
       Si manda il totale in `currency` (quello che il sito legge gia'), piu'
       la ripartizione: `currencyPurse` = addosso, `currencyBags` = per borsa. */
    const COINS = ["pp", "gp", "ep", "sp", "cp"];
    const coinsOf = (o) => {
      if (!o || typeof o !== "object") return null;
      const out = {};
      let trovate = false;
      for (const k of COINS) {
        if (o[k] !== undefined && o[k] !== null) trovate = true;
        out[k] = Math.trunc(Number(o[k]) || 0);
      }
      return trovate ? out : null;
    };
    const fsCoins = (c) => fsMap({ pp: fsInt(c.pp), gp: fsInt(c.gp), ep: fsInt(c.ep), sp: fsInt(c.sp), cp: fsInt(c.cp) });

    const tasche = coinsOf(A.currency) || { pp: 0, gp: 0, ep: 0, sp: 0, cp: 0 };
    const borse = actor.items
      .filter((i) => i.type === "container" || i.type === "backpack")
      .map((i) => ({ name: i.name, coins: coinsOf(i.system?.currency) }))
      .filter((b) => b.coins && COINS.some((k) => b.coins[k] > 0));

    const totale = {};
    for (const k of COINS) totale[k] = tasche[k] + borse.reduce((a, b) => a + b.coins[k], 0);

    const currency = fsCoins(totale);
    const currencyPurse = fsCoins(tasche);
    const currencyBags = fsArr(borse.map((b) => fsMap({
      name: fsStr(b.name),
      pp: fsInt(b.coins.pp), gp: fsInt(b.coins.gp), ep: fsInt(b.coins.ep),
      sp: fsInt(b.coins.sp), cp: fsInt(b.coins.cp),
    })));
    if (borse.length) {
      console.log(`[Sync] Monete nelle borse di ${actor.name}:`, borse.map((b) => `${b.name} ${b.coins.gp} mo`).join(" · "));
    }

    /* STATISTICHE PRINCIPALI (+ caratteristiche annidate) */
    const at = A.attributes || {};
    const stats = fsMap({
      hp: fsInt(at.hp?.value ?? 0),
      maxHp: fsInt(at.hp?.max ?? 0),
      tempHp: fsInt(at.hp?.temp ?? 0),
      ac: fsInt(at.ac?.value ?? 0),
      initiative: fsInt(at.init?.total ?? at.init?.mod ?? at.init?.bonus ?? dexMod),
      speed: fsInt(typeof at.movement?.walk === "number" ? at.movement.walk : 30),
      prof: fsInt(at.prof ?? 2),
      spellDc: fsInt(at.spelldc ?? at.spell?.dc ?? 0),
      ...abilityMap
    });

    /* PAYLOAD COMPLETO */
    const classItem = actor.itemTypes.class?.[0];
    const firestoreData = { fields: {
      name:       fsStr(actor.name),
      class:      fsStr(classItem?.name || "Viandante"),
      subclass:   fsStr(actor.itemTypes.subclass?.[0]?.name || ""),
      race:       fsStr(actor.itemTypes.race?.[0]?.name || A.details?.race || ""),
      background: fsStr(actor.itemTypes.background?.[0]?.name || A.details?.background || ""),
      image:      fsStr(actor.img),
      level:      fsInt(A.details?.level ?? 1),
      stats:      stats,
      currency:   currency,        // totale: tasche + borse
      currencyPurse: currencyPurse, // solo quello addosso
      currencyBags:  currencyBags,  // una voce per borsa che contiene monete
      spellSlots: fsArr(slots),
      skills:     fsArr(skills),
      actions:    fsArr(actions),
      inventory:  fsArr(inventory)
    }};

    const mask = ["name", "class", "subclass", "race", "background", "image", "level",
                  "stats", "currency", "currencyPurse", "currencyBags",
                  "spellSlots", "skills", "actions", "inventory"]
                 .map(f => `updateMask.fieldPaths=${f}`).join("&");

    const url = `https://firestore.googleapis.com/v1/projects/${FIREBASE_CONFIG.projectId}/databases/(default)/documents/characters/${firebaseUID}?key=${FIREBASE_CONFIG.apiKey}&${mask}`;

    try {
      const res = await fetch(url, {
        method: "PATCH",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(firestoreData)
      });
      if (res.ok) {
        const oroTxt = borse.length
          ? ` · ${totale.gp} mo (${tasche.gp} addosso, ${totale.gp - tasche.gp} in ${borse.length} ${borse.length === 1 ? "borsa" : "borse"})`
          : ` · ${totale.gp} mo`;
        ui.notifications.info(`✅ Sincronizzato: ${actor.name} → ${targetName}${oroTxt}`);
      } else {
        const err = await res.json();
        console.error("Firestore Error:", err);
        ui.notifications.error("Errore Firestore — apri la console (F12) per i dettagli.");
      }
    } catch (e) {
      console.error("Fetch Error:", e);
      ui.notifications.error("Errore di rete — apri la console (F12) per i dettagli.");
    }
  }

  run();
}
