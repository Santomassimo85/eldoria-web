// Copia di riferimento: qui SENZA la chiave API e SENZA la mappa
// giocatore→UID (repo pubblico). La copia pronta del Master sta in
// Desktop/foundry-macro-sync-all-DM.txt.
//
// Come la "Sincronizza Scheda", ma senza tendina e senza token:
// - con dei token SELEZIONATI sincronizza solo quelli;
// - senza selezione sincronizza TUTTI gli attori di tipo "character".
// Ogni attore trova il suo personaggio del sito da solo, in quest'ordine:
//   1. flag world.firebaseUID sull'attore (lo scrive ogni sync);
//   2. characters/{uid}.foundryActorId = id dell'attore;
//   3. proprietario (Owner) dell'attore che sta in uidMap;
//   4. nome dell'attore = nome del personaggio sul sito (unico).
// Se due attori puntano allo stesso personaggio si salta quello dubbio.
// Le armi accurate usano da sole la caratteristica migliore (FOR o DES).
// Prima di inviare mostra l'elenco degli abbinamenti; alla fine sussurra
// al GM il riepilogo (sincronizzati, saltati, errori).
// Scrive gli stessi campi della sync singola (stats, currency = tasche +
// borse, currencyPurse, currencyBags, inventory, actions, …) + il
// collegamento foundryActorId/foundryActorName/foundryUser e il flag.
// ================================================================

const FIREBASE_CONFIG = {
  apiKey: "TUA_API_KEY",
  projectId: "eldoria-web"
};

// Nome giocatore (come appare in Foundry) -> UID Firebase.
const uidMap = {
  // "NomeUtenteFoundry": "UID_FIREBASE",   ← compila in Foundry, non nel repo
};

/* ---- Helper: valori nel formato tipizzato di Firestore ---- */
const fsStr  = v => ({ stringValue: String(v ?? "") });
const fsInt  = v => ({ integerValue: String(Math.trunc(Number(v) || 0)) });
const fsNum  = v => ({ doubleValue: Number(v) || 0 });
const fsBool = v => ({ booleanValue: !!v });
const fsMap  = fields => ({ mapValue: { fields } });
const fsArr  = values => ({ arrayValue: { values } });

const BASE = `https://firestore.googleapis.com/v1/projects/${FIREBASE_CONFIG.projectId}/databases/(default)/documents`;

/* ---- Helper: pulisce le descrizioni dalla "roba di Foundry" ---- */
function cleanText(html) {
  if (!html) return "Nessuna descrizione.";
  let t = String(html);
  t = t.replace(/<[^>]*>/g, " ");
  t = t.replace(/&nbsp;/gi, " ").replace(/&amp;/gi, "&");
  t = t.replace(/@\w+\[([^\]|]+)(?:\|[^\]]*)?\]/g, "$1");
  t = t.replace(/&Reference\[(?:[^\]=]*=)?([^\]{}]+)\]/gi, "$1");
  t = t.replace(/\{([^}]*)\}/g, "$1");
  t = t.replace(/\[\[[^\]]*\]\]/g, "");
  t = t.replace(/\s+/g, " ").trim();
  return t || "Nessuna descrizione.";
}

/* ---- Nomi confrontabili: minuscole, niente accenti/apostrofi/virgolette ---- */
const normName = (s) => String(s || "")
  .normalize("NFD").replace(/[̀-ͯ]/g, "")
  .toLowerCase().replace(/["'’`]/g, "").replace(/\s+/g, " ").trim();

const esc = (s) => String(s ?? "").replace(/[&<>"]/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;" }[c]));

/* ---- Armi: accurata / a distanza ---- */
function isFinesse(item) {
  const props = item.system?.properties;
  if (!props) return false;
  if (props instanceof Set) return props.has("fin");
  if (Array.isArray(props)) return props.includes("fin");
  if (typeof props === "object") return props?.fin === true;
  return false;
}
function isRanged(item) {
  const actionType = item.system?.actionType;
  const props = item.system?.properties;
  if (actionType === "rwak") return true;
  if (props instanceof Set) return props.has("amm");
  if (Array.isArray(props)) return props.includes("amm");
  if (typeof props === "object") return props?.amm === true;
  return false;
}

function getDamageFormula(item, strMod, dexMod) {
  if (item.type === "spell") {
    const formula = item.labels?.damages?.[0]?.formula;
    return formula && formula !== "0" ? formula : "0";
  }
  if (item.type === "weapon") {
    const baseFormula = item.labels?.damage;
    if (!baseFormula || baseFormula === "0") return "0";
    // Accurata = la migliore fra FOR e DES (era il default della tendina).
    const mod = isFinesse(item) ? Math.max(strMod, dexMod) : isRanged(item) ? dexMod : strMod;
    if (mod !== 0) return mod > 0 ? `${baseFormula}+${mod}` : `${baseFormula}${mod}`;
    return baseFormula;
  }
  const generic = item.labels?.damage || item.labels?.damages?.[0]?.formula;
  return generic && generic !== "0" ? generic : "0";
}

/* ---- Dati di UN attore nel formato Firestore ---- */
function buildFields(actor) {
  const A = actor.system;
  const strMod = A.abilities?.str?.mod ?? 0;
  const dexMod = A.abilities?.dex?.mod ?? 0;

  const actions = actor.items
    .filter(i => ["weapon", "spell", "feat"].includes(i.type))
    .map(i => {
      const bonus = i.labels?.toHit ? i.labels.toHit.replace(/[^0-9+-]/g, "") : "0";
      let category = "Abilità";
      if (i.type === "weapon") category = "Armi";
      if (i.type === "spell") {
        const lvl = i.system.level;
        category = lvl === 0 ? "Trucchetto" : `Livello ${lvl}`;
      }
      return fsMap({
        name: fsStr(i.name),
        category: fsStr(category),
        bonus: fsStr(bonus),
        damage: fsStr(getDamageFormula(i, strMod, dexMod)),
        description: fsStr(cleanText(i.system.description?.value))
      });
    });

  const PHYS = ["weapon", "equipment", "consumable", "tool", "loot", "container", "backpack"];
  const inventory = actor.items
    .filter(i => PHYS.includes(i.type))
    .map(i => {
      const w = typeof i.system.weight === "object" ? i.system.weight?.value : i.system.weight;
      const p = typeof i.system.price === "object" ? i.system.price?.value : i.system.price;
      return fsMap({
        name: fsStr(i.name),
        type: fsStr(i.type),
        quantity: fsInt(i.system.quantity ?? 1),
        weight: fsNum(w ?? 0),
        price: fsNum(p ?? 0),
        equipped: fsBool(i.system.equipped ?? false),
        isContainer: fsBool(i.type === "container" || i.type === "backpack"),
        container: fsStr(i.system.container ?? ""),
        img: fsStr(i.img ?? ""),
        description: fsStr(cleanText(i.system.description?.value))
      });
    });

  const ab = A.abilities || {};
  const abilityMap = {};
  for (const k of ["str", "dex", "con", "int", "wis", "cha"]) {
    const a = ab[k] || {};
    const save = typeof a.save === "object" ? (a.save?.value ?? a.mod) : (a.save ?? a.mod);
    abilityMap[k] = fsMap({ score: fsInt(a.value ?? 10), mod: fsInt(a.mod ?? 0), save: fsInt(save ?? 0) });
  }

  const skills = Object.entries(A.skills || {}).map(([key, sk]) => {
    const raw = CONFIG.DND5E?.skills?.[key]?.label || key.toUpperCase();
    const label = game.i18n?.localize ? game.i18n.localize(raw) : raw;
    return fsMap({ name: fsStr(label), mod: fsInt(sk.total ?? sk.mod ?? 0), proficient: fsBool(sk.proficient) });
  });

  const sp = A.spells || {};
  const slots = [];
  for (let n = 1; n <= 9; n++) {
    const s = sp[`spell${n}`];
    if (s && (s.max ?? 0) > 0) {
      slots.push(fsMap({ label: fsStr(`Livello ${n}`), value: fsInt(s.value ?? 0), max: fsInt(s.max ?? 0) }));
    }
  }
  if (sp.pact && (sp.pact.max ?? 0) > 0) {
    slots.push(fsMap({ label: fsStr(`Patto (liv. ${sp.pact.level ?? "?"})`), value: fsInt(sp.pact.value ?? 0), max: fsInt(sp.pact.max ?? 0) }));
  }

  /* Monete: TASCHE + BORSE (i contenitori dnd5e hanno una loro currency) */
  const COINS = ["pp", "gp", "ep", "sp", "cp"];
  const coinsOf = (o) => {
    if (!o || typeof o !== "object") return null;
    const out = {};
    let trovate = false;
    for (const k of COINS) {
      if (o[k] !== undefined && o[k] !== null) trovate = true;
      out[k] = Math.trunc(Number(o[k]) || 0);
    }
    return trovate ? out : null;
  };
  const fsCoins = (c) => fsMap({ pp: fsInt(c.pp), gp: fsInt(c.gp), ep: fsInt(c.ep), sp: fsInt(c.sp), cp: fsInt(c.cp) });
  const tasche = coinsOf(A.currency) || { pp: 0, gp: 0, ep: 0, sp: 0, cp: 0 };
  const borse = actor.items
    .filter((i) => i.type === "container" || i.type === "backpack")
    .map((i) => ({ name: i.name, coins: coinsOf(i.system?.currency) }))
    .filter((b) => b.coins && COINS.some((k) => b.coins[k] > 0));
  const totale = {};
  for (const k of COINS) totale[k] = tasche[k] + borse.reduce((a, b) => a + b.coins[k], 0);

  const at = A.attributes || {};
  const stats = fsMap({
    hp: fsInt(at.hp?.value ?? 0),
    maxHp: fsInt(at.hp?.max ?? 0),
    tempHp: fsInt(at.hp?.temp ?? 0),
    ac: fsInt(at.ac?.value ?? 0),
    initiative: fsInt(at.init?.total ?? at.init?.mod ?? at.init?.bonus ?? dexMod),
    speed: fsInt(typeof at.movement?.walk === "number" ? at.movement.walk : 30),
    prof: fsInt(at.prof ?? 2),
    spellDc: fsInt(at.spelldc ?? at.spell?.dc ?? 0),
    ...abilityMap
  });

  const classItem = actor.itemTypes.class?.[0];
  const fields = {
    name:       fsStr(actor.name),
    class:      fsStr(classItem?.name || "Viandante"),
    subclass:   fsStr(actor.itemTypes.subclass?.[0]?.name || ""),
    race:       fsStr(actor.itemTypes.race?.[0]?.name || A.details?.race || ""),
    background: fsStr(actor.itemTypes.background?.[0]?.name || A.details?.background || ""),
    image:      fsStr(actor.img),
    level:      fsInt(A.details?.level ?? 1),
    stats,
    currency:      fsCoins(totale),
    currencyPurse: fsCoins(tasche),
    currencyBags:  fsArr(borse.map((b) => fsMap({
      name: fsStr(b.name),
      pp: fsInt(b.coins.pp), gp: fsInt(b.coins.gp), ep: fsInt(b.coins.ep),
      sp: fsInt(b.coins.sp), cp: fsInt(b.coins.cp),
    }))),
    spellSlots: fsArr(slots),
    skills:     fsArr(skills),
    actions:    fsArr(actions),
    inventory:  fsArr(inventory)
  };
  return { fields, gp: totale.gp, gpBags: totale.gp - tasche.gp };
}

/* ---- Personaggi del sito (lettura aperta) ---- */
async function loadSiteCharacters() {
  const out = [];
  let pageToken = "";
  do {
    const url = `${BASE}/characters?key=${FIREBASE_CONFIG.apiKey}&pageSize=300`
      + "&mask.fieldPaths=name&mask.fieldPaths=foundryActorId&mask.fieldPaths=foundryUser"
      + (pageToken ? `&pageToken=${encodeURIComponent(pageToken)}` : "");
    const res = await fetch(url);
    if (!res.ok) throw new Error(`lettura characters: HTTP ${res.status}`);
    const j = await res.json();
    for (const d of j.documents || []) {
      const f = d.fields || {};
      out.push({
        uid: d.name.split("/").pop(),
        name: f.name?.stringValue || "",
        foundryActorId: f.foundryActorId?.stringValue || "",
        foundryUser: f.foundryUser?.stringValue || ""
      });
    }
    pageToken = j.nextPageToken || "";
  } while (pageToken);
  return out;
}

/* ---- Utente Foundry proprietario dell'attore (non GM) ---- */
function ownerNames(actor) {
  const names = [];
  for (const [userId, lvl] of Object.entries(actor.ownership || {})) {
    if (userId === "default" || lvl < 3) continue;
    const u = game.users?.get(userId);
    if (u && !u.isGM) names.push(u.name);
  }
  return names;
}

/* ---- Abbina ogni attore al suo personaggio ---- */
function matchActors(actors, site) {
  const byUid = new Map(site.map((c) => [c.uid, c]));
  const userByUid = new Map(Object.entries(uidMap).map(([n, u]) => [u, n]));
  const rows = actors.map((actor) => {
    const flag = actor.getFlag("world", "firebaseUID");
    if (flag && byUid.has(flag)) return { actor, uid: flag, how: "flag" };
    const byId = site.filter((c) => c.foundryActorId === actor.id);
    if (byId.length === 1) return { actor, uid: byId[0].uid, how: "id attore" };
    const owned = ownerNames(actor).filter((n) => uidMap[n]);
    if (owned.length === 1) return { actor, uid: uidMap[owned[0]], how: `owner ${owned[0]}` };
    const n = normName(actor.name);
    const byName = site.filter((c) => c.name && normName(c.name) === n);
    if (byName.length === 1) return { actor, uid: byName[0].uid, how: "nome" };
    const why = owned.length > 1 ? `piu' owner in uidMap (${owned.join(", ")})`
      : byName.length > 1 ? "nome ambiguo sul sito" : "nessun personaggio sul sito";
    return { actor, uid: null, how: why };
  });

  // Due attori sullo stesso personaggio: vince chi e' gia' collegato
  // (flag o foundryActorId), gli altri si saltano.
  const groups = new Map();
  for (const r of rows) if (r.uid) {
    if (!groups.has(r.uid)) groups.set(r.uid, []);
    groups.get(r.uid).push(r);
  }
  for (const [uid, list] of groups) {
    if (list.length < 2) continue;
    const sure = list.filter((r) => r.how === "flag" || r.how === "id attore");
    const sameName = list.filter((r) => normName(r.actor.name) === normName(byUid.get(uid)?.name));
    const keep = sure.length === 1 ? sure[0] : sameName.length === 1 ? sameName[0] : null;
    for (const r of list) if (r !== keep) {
      r.how = `doppione di ${byUid.get(uid)?.name || uid}`;
      r.uid = null;
    }
  }
  for (const r of rows) if (r.uid) {
    r.site = byUid.get(r.uid);
    r.user = userByUid.get(r.uid) || r.site?.foundryUser || ownerNames(r.actor)[0] || "";
  }
  return rows;
}

function confirmDialog(rows) {
  const ok = rows.filter((r) => r.uid);
  const ko = rows.filter((r) => !r.uid);
  const li = ok.map((r) => `<li><strong>${esc(r.actor.name)}</strong> → ${esc(r.site?.name || r.uid)} <em style="opacity:.6">(${esc(r.how)})</em></li>`).join("");
  const skip = ko.length
    ? `<p style="margin-top:8px">Saltati (${ko.length}):</p><ul style="opacity:.7">${ko.map((r) => `<li>${esc(r.actor.name)} — ${esc(r.how)}</li>`).join("")}</ul>`
    : "";
  return new Promise((resolve) => {
    new Dialog({
      title: `Sincronizza ${ok.length} PG sul sito`,
      content: `<div style="padding:6px;max-height:420px;overflow:auto"><ul>${li || "<li>Nessun abbinamento.</li>"}</ul>${skip}</div>`,
      buttons: {
        ok: { icon: '<i class="fas fa-cloud-upload-alt"></i>', label: `Sincronizza (${ok.length})`, callback: () => resolve(ok.length > 0) },
        no: { icon: '<i class="fas fa-times"></i>', label: "Annulla", callback: () => resolve(false) }
      },
      default: "ok",
      close: () => resolve(false)
    }).render(true);
  });
}

async function syncOne(r) {
  const baseActor = game.actors.get(r.actor.id) || r.actor;
  const { fields, gp, gpBags } = buildFields(r.actor);
  fields.foundryActorId   = fsStr(baseActor.id);
  fields.foundryActorName = fsStr(baseActor.name);
  if (r.user) fields.foundryUser = fsStr(r.user);
  const mask = Object.keys(fields).map((f) => `updateMask.fieldPaths=${f}`).join("&");
  const res = await fetch(`${BASE}/characters/${r.uid}?key=${FIREBASE_CONFIG.apiKey}&${mask}`, {
    method: "PATCH",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ fields })
  });
  if (!res.ok) {
    const err = await res.json().catch(() => ({}));
    throw new Error(err?.error?.message || `HTTP ${res.status}`);
  }
  if (baseActor.getFlag("world", "firebaseUID") !== r.uid) {
    await baseActor.setFlag("world", "firebaseUID", r.uid);
  }
  return { gp, gpBags };
}

(async () => {
  if (!game.user.isGM) { ui.notifications.error("Questa macro la lancia il GM."); return; }
  if (!FIREBASE_CONFIG.apiKey || FIREBASE_CONFIG.apiKey === "TUA_API_KEY") {
    ui.notifications.error("Manca la chiave API: incolla la versione DM (foundry-macro-sync-all-DM.txt).", { permanent: true });
    return;
  }

  const selected = canvas.tokens?.controlled?.map((t) => t.actor).filter(Boolean) || [];
  const actors = selected.length
    ? [...new Map(selected.map((a) => [a.id, a])).values()]
    : game.actors.filter((a) => a.type === "character");
  if (!actors.length) { ui.notifications.warn("Nessun attore da sincronizzare."); return; }

  let site;
  try { site = await loadSiteCharacters(); }
  catch (e) { console.error(e); ui.notifications.error("Non riesco a leggere i personaggi del sito (F12 per i dettagli)."); return; }

  const rows = matchActors(actors, site);
  if (!(await confirmDialog(rows))) { ui.notifications.warn("Sincronizzazione annullata."); return; }

  const report = [];
  for (const r of rows) {
    if (!r.uid) { report.push({ attore: r.actor.name, sito: "—", esito: `saltato: ${r.how}` }); continue; }
    try {
      const { gp, gpBags } = await syncOne(r);
      report.push({ attore: r.actor.name, sito: r.site?.name || r.uid, esito: "ok", mo: gp, inBorse: gpBags });
    } catch (e) {
      console.error(`[Sync] ${r.actor.name}:`, e);
      report.push({ attore: r.actor.name, sito: r.site?.name || r.uid, esito: `errore: ${e.message}` });
    }
  }
  console.table(report);

  const ok = report.filter((x) => x.esito === "ok");
  const ko = report.filter((x) => x.esito !== "ok");
  const righe = ok.map((x) => `<li>${esc(x.attore)} · ${x.mo} mo${x.inBorse ? ` (${x.inBorse} in borsa)` : ""}</li>`).join("");
  const male = ko.map((x) => `<li style="opacity:.7">${esc(x.attore)} — ${esc(x.esito)}</li>`).join("");
  ChatMessage.create({
    whisper: ChatMessage.getWhisperRecipients("GM"),
    content: `<h3>☁ Sync sul sito: ${ok.length}/${report.length}</h3><ul>${righe}</ul>${male ? `<ul>${male}</ul>` : ""}`
  });
  ui.notifications.info(`✅ Sincronizzati ${ok.length} PG${ko.length ? ` · ${ko.length} saltati/errori (vedi chat)` : ""}.`);
})();
