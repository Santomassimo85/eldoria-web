/* ---- Helper: valori nel formato tipizzato di Firestore ---- */
const fsStr  = v => ({ stringValue: String(v ?? "") });
const fsInt  = v => ({ integerValue: String(Math.trunc(Number(v) || 0)) });
const fsNum  = v => ({ doubleValue: Number(v) || 0 });
const fsBool = v => ({ booleanValue: !!v });
const fsMap  = fields => ({ mapValue: { fields } });
const fsArr  = values => ({ arrayValue: { values } });

const BASE = `https://firestore.googleapis.com/v1/projects/${FIREBASE_CONFIG.projectId}/databases/(default)/documents`;

/* ---- Helper: pulisce le descrizioni dalla "roba di Foundry" ---- */
function cleanText(html) {
  if (!html) return "Nessuna descrizione.";
  let t = String(html);
  t = t.replace(/<[^>]*>/g, " ");
  t = t.replace(/&nbsp;/gi, " ").replace(/&amp;/gi, "&");
  t = t.replace(/@\w+\[([^\]|]+)(?:\|[^\]]*)?\]/g, "$1");
  t = t.replace(/&Reference\[(?:[^\]=]*=)?([^\]{}]+)\]/gi, "$1");
  t = t.replace(/\{([^}]*)\}/g, "$1");
  t = t.replace(/\[\[[^\]]*\]\]/g, "");
  t = t.replace(/\s+/g, " ").trim();
  return t || "Nessuna descrizione.";
}

/* ---- Nomi confrontabili: minuscole, niente accenti/apostrofi/virgolette ---- */
const normName = (s) => String(s || "")
  .normalize("NFD").replace(/[̀-ͯ]/g, "")
  .toLowerCase().replace(/["'’`]/g, "").replace(/\s+/g, " ").trim();

const esc = (s) => String(s ?? "").replace(/[&<>"]/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;" }[c]));

/* ---- Armi: accurata / a distanza ---- */
function isFinesse(item) {
  const props = item.system?.properties;
  if (!props) return false;
  if (props instanceof Set) return props.has("fin");
  if (Array.isArray(props)) return props.includes("fin");
  if (typeof props === "object") return props?.fin === true;
  return false;
}
function isRanged(item) {
  const actionType = item.system?.actionType;
  const props = item.system?.properties;
  if (actionType === "rwak") return true;
  if (props instanceof Set) return props.has("amm");
  if (Array.isArray(props)) return props.includes("amm");
  if (typeof props === "object") return props?.amm === true;
  return false;
}

function getDamageFormula(item, strMod, dexMod) {
  if (item.type === "spell") {
    const formula = item.labels?.damages?.[0]?.formula;
    return formula && formula !== "0" ? formula : "0";
  }
  if (item.type === "weapon") {
    const baseFormula = item.labels?.damage;
    if (!baseFormula || baseFormula === "0") return "0";
    // Accurata = la migliore fra FOR e DES (era il default della tendina).
    const mod = isFinesse(item) ? Math.max(strMod, dexMod) : isRanged(item) ? dexMod : strMod;
    if (mod !== 0) return mod > 0 ? `${baseFormula}+${mod}` : `${baseFormula}${mod}`;
    return baseFormula;
  }
  const generic = item.labels?.damage || item.labels?.damages?.[0]?.formula;
  return generic && generic !== "0" ? generic : "0";
}

/* ---- Dati di UN attore nel formato Firestore ---- */
function buildFields(actor) {
  const A = actor.system;
  const strMod = A.abilities?.str?.mod ?? 0;
  const dexMod = A.abilities?.dex?.mod ?? 0;

  const actions = actor.items
    .filter(i => ["weapon", "spell", "feat"].includes(i.type))
    .map(i => {
      const bonus = i.labels?.toHit ? i.labels.toHit.replace(/[^0-9+-]/g, "") : "0";
      let category = "Abilità";
      if (i.type === "weapon") category = "Armi";
      if (i.type === "spell") {
        const lvl = i.system.level;
        category = lvl === 0 ? "Trucchetto" : `Livello ${lvl}`;
      }
      return fsMap({
        name: fsStr(i.name),
        category: fsStr(category),
        bonus: fsStr(bonus),
        damage: fsStr(getDamageFormula(i, strMod, dexMod)),
        description: fsStr(cleanText(i.system.description?.value))
      });
    });

  const PHYS = ["weapon", "equipment", "consumable", "tool", "loot", "container", "backpack"];
  const inventory = actor.items
    .filter(i => PHYS.includes(i.type))
    .map(i => {
      const w = typeof i.system.weight === "object" ? i.system.weight?.value : i.system.weight;
      const p = typeof i.system.price === "object" ? i.system.price?.value : i.system.price;
      return fsMap({
        name: fsStr(i.name),
        type: fsStr(i.type),
        quantity: fsInt(i.system.quantity ?? 1),
        weight: fsNum(w ?? 0),
        price: fsNum(p ?? 0),
        equipped: fsBool(i.system.equipped ?? false),
        isContainer: fsBool(i.type === "container" || i.type === "backpack"),
        container: fsStr(i.system.container ?? ""),
        img: fsStr(i.img ?? ""),
        description: fsStr(cleanText(i.system.description?.value))
      });
    });

  const ab = A.abilities || {};
  const abilityMap = {};
  for (const k of ["str", "dex", "con", "int", "wis", "cha"]) {
    const a = ab[k] || {};
    const save = typeof a.save === "object" ? (a.save?.value ?? a.mod) : (a.save ?? a.mod);
    abilityMap[k] = fsMap({ score: fsInt(a.value ?? 10), mod: fsInt(a.mod ?? 0), save: fsInt(save ?? 0) });
  }

  const skills = Object.entries(A.skills || {}).map(([key, sk]) => {
    const raw = CONFIG.DND5E?.skills?.[key]?.label || key.toUpperCase();
    const label = game.i18n?.localize ? game.i18n.localize(raw) : raw;
    return fsMap({ name: fsStr(label), mod: fsInt(sk.total ?? sk.mod ?? 0), proficient: fsBool(sk.proficient) });
  });

  const sp = A.spells || {};
  const slots = [];
  for (let n = 1; n <= 9; n++) {
    const s = sp[`spell${n}`];
    if (s && (s.max ?? 0) > 0) {
      slots.push(fsMap({ label: fsStr(`Livello ${n}`), value: fsInt(s.value ?? 0), max: fsInt(s.max ?? 0) }));
    }
  }
  if (sp.pact && (sp.pact.max ?? 0) > 0) {
    slots.push(fsMap({ label: fsStr(`Patto (liv. ${sp.pact.level ?? "?"})`), value: fsInt(sp.pact.value ?? 0), max: fsInt(sp.pact.max ?? 0) }));
  }

  /* Monete: TASCHE + BORSE (i contenitori dnd5e hanno una loro currency) */
  const COINS = ["pp", "gp", "ep", "sp", "cp"];
  const coinsOf = (o) => {
    if (!o || typeof o !== "object") return null;
    const out = {};
    let trovate = false;
    for (const k of COINS) {
      if (o[k] !== undefined && o[k] !== null) trovate = true;
      out[k] = Math.trunc(Number(o[k]) || 0);
    }
    return trovate ? out : null;
  };
  const fsCoins = (c) => fsMap({ pp: fsInt(c.pp), gp: fsInt(c.gp), ep: fsInt(c.ep), sp: fsInt(c.sp), cp: fsInt(c.cp) });
  const tasche = coinsOf(A.currency) || { pp: 0, gp: 0, ep: 0, sp: 0, cp: 0 };
  const borse = actor.items
    .filter((i) => i.type === "container" || i.type === "backpack")
    .map((i) => ({ name: i.name, coins: coinsOf(i.system?.currency) }))
    .filter((b) => b.coins && COINS.some((k) => b.coins[k] > 0));
  const totale = {};
  for (const k of COINS) totale[k] = tasche[k] + borse.reduce((a, b) => a + b.coins[k], 0);

  const at = A.attributes || {};
  const stats = fsMap({
    hp: fsInt(at.hp?.value ?? 0),
    maxHp: fsInt(at.hp?.max ?? 0),
    tempHp: fsInt(at.hp?.temp ?? 0),
    ac: fsInt(at.ac?.value ?? 0),
    initiative: fsInt(at.init?.total ?? at.init?.mod ?? at.init?.bonus ?? dexMod),
    speed: fsInt(typeof at.movement?.walk === "number" ? at.movement.walk : 30),
    prof: fsInt(at.prof ?? 2),
    spellDc: fsInt(at.spelldc ?? at.spell?.dc ?? 0),
    ...abilityMap
  });

  const classItem = actor.itemTypes.class?.[0];
  const fields = {
    name:       fsStr(actor.name),
    class:      fsStr(classItem?.name || "Viandante"),
    subclass:   fsStr(actor.itemTypes.subclass?.[0]?.name || ""),
    race:       fsStr(actor.itemTypes.race?.[0]?.name || A.details?.race || ""),
    background: fsStr(actor.itemTypes.background?.[0]?.name || A.details?.background || ""),
    image:      fsStr(actor.img),
    level:      fsInt(A.details?.level ?? 1),
    stats,
    currency:      fsCoins(totale),
    currencyPurse: fsCoins(tasche),
    currencyBags:  fsArr(borse.map((b) => fsMap({
      name: fsStr(b.name),
      pp: fsInt(b.coins.pp), gp: fsInt(b.coins.gp), ep: fsInt(b.coins.ep),
      sp: fsInt(b.coins.sp), cp: fsInt(b.coins.cp),
    }))),
    spellSlots: fsArr(slots),
    skills:     fsArr(skills),
    actions:    fsArr(actions),
    inventory:  fsArr(inventory)
  };
  return { fields, gp: totale.gp, gpBags: totale.gp - tasche.gp };
}

/* ---- Personaggi del sito (lettura aperta) ---- */
async function loadSiteCharacters() {
  const out = [];
  let pageToken = "";
  do {
    const url = `${BASE}/characters?key=${FIREBASE_CONFIG.apiKey}&pageSize=300`
      + "&mask.fieldPaths=name&mask.fieldPaths=foundryActorId&mask.fieldPaths=foundryUser"
      + (pageToken ? `&pageToken=${encodeURIComponent(pageToken)}` : "");
    const res = await fetch(url);
    if (!res.ok) throw new Error(`lettura characters: HTTP ${res.status}`);
    const j = await res.json();
    for (const d of j.documents || []) {
      const f = d.fields || {};
      out.push({
        uid: d.name.split("/").pop(),
        name: f.name?.stringValue || "",
        foundryActorId: f.foundryActorId?.stringValue || "",
        foundryUser: f.foundryUser?.stringValue || ""
      });
    }
    pageToken = j.nextPageToken || "";
  } while (pageToken);
  return out;
}

/* ---- Nomi utente dei proprietari (non GM) ---- */
function ownerNames(actor) {
  const names = [];
  for (const [userId, lvl] of Object.entries(actor.ownership || {})) {
    if (userId === "default" || lvl < 3) continue;
    const u = game.users?.get(userId);
    if (u && !u.isGM) names.push(u.name);
  }
  return names;
}

/* ---- Attore del personaggio ---- */
// flag world.firebaseUID → foundryActorId → personaggio assegnato all'utente
// → unico attore posseduto dall'utente (o quello con lo stesso nome) → nome.
function actorFor(c, user, chars) {
  const flagged = chars.filter((a) => a.getFlag("world", "firebaseUID") === c.uid);
  if (flagged.length === 1) return { actor: flagged[0], how: "flag" };
  if (flagged.length > 1) {
    const pick = flagged.find((a) => a.id === c.foundryActorId) || flagged.find((a) => normName(a.name) === normName(c.name));
    if (pick) return { actor: pick, how: "flag (doppione risolto)" };
  }
  const byId = c.foundryActorId ? game.actors.get(c.foundryActorId) : null;
  if (byId && byId.type === "character") return { actor: byId, how: "id attore" };
  if (user?.character && user.character.type === "character") return { actor: user.character, how: `assegnato a ${user.name}` };
  if (user) {
    const owned = chars.filter((a) => (a.ownership?.[user.id] ?? 0) >= 3);
    if (owned.length === 1) return { actor: owned[0], how: `posseduto da ${user.name}` };
    const same = owned.filter((a) => normName(a.name) === normName(c.name));
    if (same.length === 1) return { actor: same[0], how: `posseduto da ${user.name}` };
  }
  const byName = chars.filter((a) => normName(a.name) === normName(c.name));
  if (byName.length === 1) return { actor: byName[0], how: "nome" };
  return { actor: null, how: byName.length > 1 ? "più attori con lo stesso nome" : "nessun attore trovato" };
}

/* ---- Piano: per ogni PG attivo, attore + cosa va sistemato ---- */
function makePlan(active) {
  const chars = game.actors.filter((a) => a.type === "character");
  const used = new Map();
  const rows = active.map((c) => {
    const user = userFor(c);
    const { actor, how } = actorFor(c, user, chars);
    const fix = [];
    if (actor) {
      if (actor.getFlag("world", "firebaseUID") !== c.uid) fix.push("collegamento al sito");
      if (user && (actor.ownership?.[user.id] ?? 0) < 3) fix.push(`${user.name} proprietario`);
      if (user && user.character?.id !== actor.id) fix.push(`assegnato a ${user.name}`);
      const stale = chars.filter((a) => a.id !== actor.id && a.getFlag("world", "firebaseUID") === c.uid);
      if (stale.length) fix.push(`tolto il collegamento da ${stale.map((a) => a.name).join(", ")}`);
      if (used.has(actor.id)) return { c, user, actor: null, how: `attore già usato da ${used.get(actor.id)}`, fix: [] };
      used.set(actor.id, c.name);
      return { c, user, actor, how, fix, stale };
    }
    return { c, user, actor: null, how, fix: [] };
  });
  return rows;
}

function val(v) {
  if (v.stringValue !== undefined) return v.stringValue;
  if (v.integerValue !== undefined) return Number(v.integerValue);
  if (v.doubleValue !== undefined) return Number(v.doubleValue);
  if (v.booleanValue !== undefined) return v.booleanValue;
  if (v.nullValue !== undefined) return null;
  if (v.timestampValue !== undefined) return v.timestampValue;
  if (v.mapValue) return obj(v.mapValue.fields || {});
  if (v.arrayValue) return (v.arrayValue.values || []).map(val);
  return null;
}
function obj(fields) {
  const o = {};
  for (const [k, v] of Object.entries(fields)) o[k] = val(v);
  return o;
}

const DND_MAJOR = parseInt((game.system.version || "3").split(".")[0]) || 3;

function buildItemData(d) {
  const descHtml = /<[a-z][\s\S]*>/i.test(d.description || "")
    ? (d.description || "")
    : `<p>${(d.description || "").replace(/\n/g, "<br>")}</p>`;
  const system = {
    description: { value: descHtml, chat: "" },
    quantity: d.quantity || 1,
    price: { value: d.price || 0, denomination: "gp" },
    rarity: d.rarity || "",
    identified: true,
  };
  system.weight = DND_MAJOR >= 3 ? { value: d.weight || 0, units: "lb" } : (d.weight || 0);
  const parts = [];
  if (d.damageFormula) parts.push([d.damageFormula, d.damageType || ""]);
  if (d.damage2Formula) parts.push([d.damage2Formula, d.damage2Type || ""]);
  if (parts.length || d.versatileFormula) system.damage = { parts, versatile: d.versatileFormula || "" };
  if (d.actionType) system.actionType = d.actionType;
  if (d.attackBonus) system.attackBonus = String(d.attackBonus);
  if (d.saveAbility && d.saveDC) system.save = { ability: d.saveAbility, dc: d.saveDC, scaling: "flat" };
  if (d.foundryType === "weapon") {
    system.proficient = d.proficient ? 1 : 0;
    const props = Array.isArray(d.properties) ? d.properties : [];
    system.properties = DND_MAJOR >= 3 ? props : props.reduce((o, k) => (o[k] = true, o), {});
  }
  if (d.foundryType === "equipment" && d.armorType) {
    system.armor = { value: d.armorValue || 0, type: d.armorType, dex: null };
    system.type = { value: d.armorType };
  }
  const out = { name: d.name || "Oggetto", type: d.foundryType || "loot", system };
  if (d.img) out.img = d.img;
  return out;
}

}

// Dopo aver scalato su Foundry: oro nuovo sul sito + spesa tolta da goldPending.
async function aggiornaSito(uid, gpDopo, scalaPending) {
  const nome = `projects/${FIREBASE_CONFIG.projectId}/databases/(default)/documents/characters/${uid}`;
  const write = {
    update: { name: nome, fields: { currency: { mapValue: { fields: { gp: { integerValue: String(Math.max(0, Math.round(gpDopo))) } } } } } },
    updateMask: { fieldPaths: ["currency.gp"] },
    currentDocument: { exists: true },
  };
  if (scalaPending > 0) {
    write.updateTransforms = [{ fieldPath: "crafting.goldPending", increment: { integerValue: String(-Math.round(scalaPending)) } }];
  }
  const res = await fetch(`https://firestore.googleapis.com/v1/projects/${FIREBASE_CONFIG.projectId}/databases/(default)/documents:commit?key=${FIREBASE_CONFIG.apiKey}`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ writes: [write] }),
  });
  if (!res.ok) throw new Error(`sito non aggiornato (${res.status}): ${await res.text()}`);
}

const cancellaVoce = (d) => fetch(`https://firestore.googleapis.com/v1/${d._path}?key=${FIREBASE_CONFIG.apiKey}`, { method: "DELETE" });

async function scalaMonete(actor, d) {
  const amount = Math.max(0, Number(d.amount) || 0);
  const add = d.op === "add";
  const borse = actor.items.filter((i) =>
    (i.type === "container" || i.type === "backpack") && foundry.utils.getProperty(i, "system.currency"));
  const casse = [
    { nome: "tasche", leggi: () => Number(foundry.utils.getProperty(actor, "system.currency.gp")) || 0,
      scrivi: (v) => actor.update({ "system.currency.gp": v }) },
    ...borse.map((b) => ({ nome: b.name,
      leggi: () => Number(foundry.utils.getProperty(b, "system.currency.gp")) || 0,
      scrivi: (v) => b.update({ "system.currency.gp": v }) })),
  ];
  if (!PRIMA_LE_TASCHE) casse.reverse();
  const prima = casse.reduce((a, c) => a + c.leggi(), 0);
  const mosse = [];
  let resto = amount;
  if (add) {
    // le monete in più vanno sempre addosso al PG
    const t = casse.find((c) => c.nome === "tasche") || casse[0];
    await t.scrivi(t.leggi() + amount);
    mosse.push(`${t.nome} +${amount}`);
    resto = 0;
  } else {
    for (const c of casse) {
      if (resto <= 0) break;
      const ha = c.leggi();
      if (ha <= 0) continue;
      const preso = Math.min(ha, resto);
      await c.scrivi(ha - preso);
      resto -= preso;
      mosse.push(`${c.nome} −${preso}`);
    }
  }
  const dopo = casse.reduce((a, c) => a + c.leggi(), 0);
  await ChatMessage.create({
    speaker: { alias: "Officina" },
    content: `<p><strong>${esc(actor.name)}</strong>: ${add ? "+" : "−"}${amount} mo (${prima} → ${dopo}).</p>`
      + (mosse.length > 1 ? `<p style="font-size:.9em;opacity:.8">Presi da: ${esc(mosse.join(", "))}.</p>` : "")
      + `<p style="font-size:.9em;opacity:.8">${esc(d.reason || "")}</p>`
      + (resto ? `<p style="color:#b91c1c"><strong>Attenzione:</strong> gli mancavano ${resto} mo, tasche e borse sono a 0.</p>` : ""),
    whisper: ChatMessage.getWhisperRecipients("GM").map((u) => u.id),
  });
  // Il sito si allinea subito. Se non ci riesce la voce si cancella lo stesso:
  // l'oro su Foundry è già scalato e rifarlo sarebbe peggio (lo sistema la sync).
  try { await aggiornaSito(d.targetUid, dopo, d.pendingOnSite && !add ? amount : 0); }
  catch (e) { console.warn("[Coda] oro scalato su Foundry ma il sito non si è aggiornato:", e); }
}

// Processa la coda. attoreDi(uid) → attore del PG (dal piano dei collegamenti).
// Ritorna { oggetti, spese, saltate, toccati:Set(uid), righe:[] }.
async function processaCoda(voci, attoreDi) {
  const esito = { oggetti: 0, spese: 0, saltate: 0, toccati: new Set(), righe: [] };
  for (const d of voci) {
    try {
      if (d.kind === "gold" || d.target === "player") {
        const actor = attoreDi(d.targetUid);
        if (!actor) {
          esito.saltate++;
          esito.righe.push(`⏸ ${descriviVoce(d)} — PG senza attore, resta in coda`);
          continue;
        }
        if (d.kind === "gold") { await scalaMonete(actor, d); esito.spese++; }
        else { await actor.createEmbeddedDocuments("Item", [buildItemData(d)]); esito.oggetti++; }
        esito.toccati.add(d.targetUid);
      } else {
        await Item.create(buildItemData(d));
        esito.oggetti++;
      }
      await cancellaVoce(d);
      esito.righe.push(`✓ ${descriviVoce(d)}`);
    } catch (e) {
      console.error("[Coda]", d, e);
      esito.saltate++;
      esito.righe.push(`✗ ${descriviVoce(d)} — ${e.message || e}`);
    }
  }
  return esito;
}

// Attore di un UID qualsiasi (anche di un PG non attivo che ha roba in coda).
function attoreDaFlag(uid) {
  return game.actors.find((a) => a.type === "character" && a.getFlag("world", "firebaseUID") === uid) || null;
}

function confirmDialog(rows, fonte, coda) {
  const ok = rows.filter((r) => r.actor);
  const ko = rows.filter((r) => !r.actor);
  const li = ok.map((r) => `<li><strong>${esc(r.c.name)}</strong> ← ${esc(r.actor.name)} · ${esc(r.user?.name || "nessun utente")}`
    + ` <em style="opacity:.6">(${esc(r.how)})</em>`
    + (r.fix.length ? `<br><span style="color:#c07b00">da sistemare: ${esc(r.fix.join(" · "))}</span>` : "")
    + `</li>`).join("");
  const skip = ko.length
    ? `<p style="margin-top:8px;color:#b00">Non trovati (${ko.length}):</p><ul>${ko.map((r) => `<li>${esc(r.c.name)} — ${esc(r.how)}</li>`).join("")}</ul>`
    : "";
  const codaHtml = coda.errore
    ? `<p style="color:#b00">Coda dell'app non leggibile: ${esc(coda.errore)}</p>`
    : coda.voci.length
      ? `<ul>${coda.voci.map((d) => `<li>${esc(descriviVoce(d))}</li>`).join("")}</ul>`
      : `<p style="opacity:.6">Niente in coda.</p>`;
  return new Promise((resolve) => {
    new Dialog({
      title: `Collega, porta il crafting e sincronizza (${ok.length} PG)`,
      content: `<div style="padding:6px;max-height:460px;overflow:auto">`
        + `<h3 style="margin:0 0 4px">1 · PG attivi (${esc(fonte)})</h3><ul>${li || "<li>Nessuno.</li>"}</ul>${skip}`
        + `<h3 style="margin:10px 0 4px">2 · Dall'app a Foundry (${coda.voci.length})</h3>${codaHtml}`
        + `<h3 style="margin:10px 0 4px">3 · Schede, monete e inventario → sito</h3><p style="opacity:.7">Dopo il passo 2, così sul sito risultano già i soldi scalati e gli oggetti nuovi.</p>`
        + `</div>`,
      buttons: {
        ok: { icon: '<i class="fas fa-cloud-upload-alt"></i>', label: "Vai", callback: () => resolve(true) },
        no: { icon: '<i class="fas fa-times"></i>', label: "Annulla", callback: () => resolve(false) }
      },
      default: "ok",
      close: () => resolve(false)
    }).render(true);
  });
}

/* ---- Collega attore ↔ giocatore ↔ sito ---- */
async function linkOne(r) {
  const base = game.actors.get(r.actor.id) || r.actor;
  if (base.getFlag("world", "firebaseUID") !== r.c.uid) await base.setFlag("world", "firebaseUID", r.c.uid);
  if (r.user && (base.ownership?.[r.user.id] ?? 0) < 3) await base.update({ [`ownership.${r.user.id}`]: 3 });
  if (r.user && r.user.character?.id !== base.id) await r.user.update({ character: base.id });
  for (const a of r.stale || []) await a.unsetFlag("world", "firebaseUID");
}

/* ---- Scheda, monete, inventario → sito ---- */
async function syncOne(uid, actor, userName) {
  const base = game.actors.get(actor.id) || actor;
  const { fields, gp, gpBags } = buildFields(base);
  fields.foundryActorId   = fsStr(base.id);
  fields.foundryActorName = fsStr(base.name);
  if (userName) fields.foundryUser = fsStr(userName);
  const mask = Object.keys(fields).map((f) => `updateMask.fieldPaths=${f}`).join("&");
  const res = await fetch(`${BASE}/characters/${uid}?key=${FIREBASE_CONFIG.apiKey}&${mask}`, {
    method: "PATCH",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ fields })
  });
  if (!res.ok) {
    const err = await res.json().catch(() => ({}));
    throw new Error(err?.error?.message || `HTTP ${res.status}`);
  }
  return { gp, gpBags };
}

(async () => {
  if (!game.user.isGM) { ui.notifications.error("Questa macro la lancia il GM."); return; }
  if (!FIREBASE_CONFIG.apiKey || FIREBASE_CONFIG.apiKey === "TUA_API_KEY") {
    ui.notifications.error("Manca la chiave API: incolla la versione DM (foundry-macro-sync-all-DM.txt).", { permanent: true });
    return;
  }
  // Un solo giro alla volta, e un solo timer (spegne anche quello della vecchia "itemsFromApp").
  const G = (window.__eldoriaImport ||= { timer: null, inCorso: false });
  if (G.inCorso) { ui.notifications.warn("C'è già un giro in corso: aspetta che finisca."); return; }

  let site;
  try { site = await loadSiteCharacters(); }
  catch (e) { console.error(e); ui.notifications.error("Non riesco a leggere i personaggi del sito (F12 per i dettagli)."); return; }
  const hidden = await loadHidden();
  const active = activeCharacters(site, hidden);
  const rows = makePlan(active);
  const coda = { voci: [], errore: "" };
  try { coda.voci = await leggiCoda(); } catch (e) { coda.errore = e.message; }

  if (!(await confirmDialog(rows, hidden ? "lista dell'app" : "lista di riserva della macro", coda))) {
    ui.notifications.warn("Annullato.");
    return;
  }

  G.inCorso = true;
  const report = [];
  let esitoCoda = null;
  try {
    // 1) COLLEGA
    for (const r of rows) {
      if (!r.actor) continue;
      try { await linkOne(r); } catch (e) { console.error(`[Collega] ${r.c.name}:`, e); r.linkErr = e.message; }
    }
    // 2) CODA DELL'APP → FOUNDRY (oggetti e monete del crafting)
    const perUid = new Map(rows.filter((r) => r.actor).map((r) => [r.c.uid, r.actor]));
    const attoreDi = (uid) => perUid.get(uid) || attoreDaFlag(uid);
    if (coda.voci.length) esitoCoda = await processaCoda(coda.voci, attoreDi);
    // 3) SCHEDE → SITO
    for (const r of rows) {
      if (!r.actor) { report.push({ pg: r.c.name, esito: `non trovato: ${r.how}` }); continue; }
      try {
        const { gp, gpBags } = await syncOne(r.c.uid, r.actor, r.user?.name);
        report.push({ pg: r.c.name, esito: r.linkErr ? `ok (collegamento: ${r.linkErr})` : "ok", mo: gp, inBorse: gpBags, sistemato: r.fix.join(" · ") });
      } catch (e) {
        console.error(`[Sync] ${r.c.name}:`, e);
        report.push({ pg: r.c.name, esito: `errore: ${e.message}` });
      }
    }
  } finally { G.inCorso = false; }
  console.table(report);

  const ok = report.filter((x) => x.esito.startsWith("ok"));
  const ko = report.filter((x) => !x.esito.startsWith("ok"));
  const righe = ok.map((x) => `<li>${esc(x.pg)} · ${x.mo} mo${x.inBorse ? ` (${x.inBorse} in borsa)` : ""}${x.sistemato ? ` <em>— ${esc(x.sistemato)}</em>` : ""}</li>`).join("");
  const male = ko.map((x) => `<li style="opacity:.7">${esc(x.pg)} — ${esc(x.esito)}</li>`).join("");
  const codaRighe = esitoCoda ? `<p><strong>Dall'app:</strong> ${esitoCoda.oggetti} oggetti, ${esitoCoda.spese} spese${esitoCoda.saltate ? `, ${esitoCoda.saltate} rimaste in coda` : ""}</p><ul>${esitoCoda.righe.map((t) => `<li>${esc(t)}</li>`).join("")}</ul>` : "";
  ChatMessage.create({
    whisper: ChatMessage.getWhisperRecipients("GM"),
    content: `<h3>☁ Sito ↔ Foundry</h3>${codaRighe}<p><strong>Schede sul sito:</strong> ${ok.length}/${report.length}</p><ul>${righe}</ul>${male ? `<ul>${male}</ul>` : ""}`
  });
  ui.notifications.info(`✅ ${ok.length} PG sincronizzati${esitoCoda ? ` · dall'app ${esitoCoda.oggetti} oggetti e ${esitoCoda.spese} spese` : ""}${ko.length ? ` · ${ko.length} con problemi (vedi chat)` : ""}.`);

  // 4) CODA IN AUTOMATICO: finché Foundry resta aperto, ogni CODA_OGNI_MINUTI
  //    porta da solo oggetti e spese nuove dell'Officina e aggiorna sul sito
  //    la scheda dei PG toccati. Niente finestre, niente doppioni.
  if (G.timer) { clearInterval(G.timer); G.timer = null; }
  if (CODA_OGNI_MINUTI > 0) {
    const nomeUtente = new Map(rows.filter((r) => r.actor).map((r) => [r.c.uid, r.user?.name]));
    const perUid = new Map(rows.filter((r) => r.actor).map((r) => [r.c.uid, r.actor]));
    G.timer = setInterval(async () => {
      if (G.inCorso) return;
      G.inCorso = true;
      try {
        const voci = await leggiCoda();
        if (!voci.length) return;
        const es = await processaCoda(voci, (uid) => perUid.get(uid) || attoreDaFlag(uid));
        for (const uid of es.toccati) {
          const a = perUid.get(uid) || attoreDaFlag(uid);
          if (a) await syncOne(uid, a, nomeUtente.get(uid)).catch((e) => console.warn("[Coda] sync:", e));
        }
        if (es.oggetti || es.spese) {
          ChatMessage.create({ whisper: ChatMessage.getWhisperRecipients("GM"), content: `<p>🔁 Dall'app: ${es.oggetti} oggetti, ${es.spese} spese.</p><ul>${es.righe.map((t) => `<li>${esc(t)}</li>`).join("")}</ul>` });
        }
      } catch (e) { console.warn("[Coda] giro automatico:", e); }
      finally { G.inCorso = false; }
    }, CODA_OGNI_MINUTI * 60 * 1000);
    ui.notifications.info(`🔁 Officina automatica accesa: ogni ${CODA_OGNI_MINUTI} minuti porto oggetti e spese finché Foundry resta aperto.`);
  }
})();
