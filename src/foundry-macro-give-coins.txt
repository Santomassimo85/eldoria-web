// =============================================================
// DAI MONETE AI PG (D&D 5e) — macro del GM (2026-10-25)
// Una casella per ogni moneta (mp, mo, me, ma, mr): lasci 0 quelle che non
// servono. A chi: tutti i PG spuntati di default, oppure scegli tu.
// Le monete vanno ADDOSSO al PG (system.currency dell'attore).
// Numero negativo = togli monete (mai sotto lo 0). Riepilogo in chat.
// Il sito vede le monete nuove al prossimo "Fetch admin" / "fetch".
// =============================================================

(async () => {
  if (!game.user.isGM) { ui.notifications.error("Questa macro la lancia il GM."); return; }

  const MONETE = [
    { key: "pp", label: "💎 Platino", short: "mp" },
    { key: "gp", label: "🟡 Oro", short: "mo" },
    { key: "ep", label: "🔵 Electrum", short: "me" },
    { key: "sp", label: "⚪ Argento", short: "ma" },
    { key: "cp", label: "🟤 Rame", short: "mr" }
  ];
  const esc = (s) => String(s ?? "").replace(/[&<>"]/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;" }[c]));
  const borsa = (pg) => {
    const c = pg.system?.currency || {};
    return MONETE.filter((m) => (c[m.key] ?? 0) || ["gp", "sp", "cp"].includes(m.key))
      .map((m) => `${c[m.key] ?? 0} ${m.short}`).join(" · ");
  };

  const pgs = game.actors.filter((a) => a.type === "character" && a.hasPlayerOwner)
    .sort((a, b) => a.name.localeCompare(b.name));
  if (!pgs.length) { ui.notifications.warn("Nessun PG trovato."); return; }

  const caselle = MONETE.map((m) => `
    <div style="flex:1;min-width:70px;">
      <label style="font-size:.85em;"><b>${m.label}</b></label>
      <input type="number" name="c_${m.key}" value="0" step="1" style="width:100%;margin-top:2px;"/>
    </div>`).join("");

  const lista = pgs.map((pg) => `
    <label style="display:block;margin:4px 0;cursor:pointer;">
      <input type="checkbox" name="pg" value="${pg.id}" checked/>
      ${esc(pg.name)} <span style="color:#888;font-size:.85em;">(${borsa(pg)})</span>
    </label>`).join("");

  const content = `
    <form style="padding:4px;">
      <b>Quante a testa</b>
      <div style="display:flex;gap:6px;flex-wrap:wrap;margin:6px 0;">${caselle}</div>
      <p style="color:#888;font-size:.85em;margin:0 0 6px;">0 = quella moneta non cambia. Numero negativo = togli (mai sotto lo 0).</p>
      <hr>
      <b>A chi:</b>
      <div style="margin:6px 0;">
        <button type="button" data-azione="tutti" style="width:auto;margin-right:4px;">Seleziona tutti</button>
        <button type="button" data-azione="nessuno" style="width:auto;">Deseleziona tutti</button>
      </div>
      <div style="max-height:240px;overflow-y:auto;border:1px solid #ccc;padding:8px;border-radius:4px;">${lista}</div>
    </form>`;

  // html può essere jQuery (Foundry ≤ v12) o un elemento (v13): prendo sempre il DOM
  const domDi = (html) => (html?.[0] ?? html);

  const scelta = await new Promise((resolve) => {
    new Dialog({
      title: "💰 Dai monete",
      content,
      buttons: {
        ok: {
          icon: '<i class="fas fa-coins"></i>',
          label: "Assegna",
          callback: (html) => {
            const root = domDi(html);
            const importi = {};
            MONETE.forEach((m) => {
              const n = parseInt(root.querySelector(`[name="c_${m.key}"]`).value, 10);
              if (Number.isFinite(n) && n !== 0) importi[m.key] = n;
            });
            resolve({ importi, ids: [...root.querySelectorAll('input[name="pg"]:checked')].map((i) => i.value) });
          }
        },
        no: { icon: '<i class="fas fa-times"></i>', label: "Annulla", callback: () => resolve(null) }
      },
      default: "ok",
      render: (html) => {
        const root = domDi(html);
        root.querySelectorAll("[data-azione]").forEach((b) => b.addEventListener("click", (ev) => {
          ev.preventDefault();
          const on = b.dataset.azione === "tutti";
          root.querySelectorAll('input[name="pg"]').forEach((c) => { c.checked = on; });
        }));
      },
      close: () => resolve(null)
    }).render(true);
  });

  if (!scelta) return;
  const { importi, ids } = scelta;
  const chiavi = Object.keys(importi);
  if (!chiavi.length) { ui.notifications.warn("Tutte le monete sono a 0: niente da fare."); return; }
  if (!ids.length) { ui.notifications.warn("Nessun PG selezionato."); return; }

  const righe = [];
  for (const id of ids) {
    const pg = game.actors.get(id);
    if (!pg) continue;
    const update = {};
    const pezzi = [];
    let scoperto = false;
    for (const k of chiavi) {
      const prima = Number(foundry.utils.getProperty(pg, `system.currency.${k}`)) || 0;
      const dopo = Math.max(0, prima + importi[k]);
      if (prima + importi[k] < 0) scoperto = true;
      update[`system.currency.${k}`] = dopo;
      pezzi.push(`${MONETE.find((m) => m.key === k).short} ${prima} → ${dopo}`);
    }
    try {
      await pg.update(update);
      righe.push(`<li><b>${esc(pg.name)}</b>: ${pezzi.join(" · ")}${scoperto ? " ⚠ non bastavano" : ""}</li>`);
    } catch (e) {
      console.error(e);
      righe.push(`<li><b>${esc(pg.name)}</b>: ❌ errore (F12)</li>`);
    }
  }

  const titolo = chiavi.map((k) => `${importi[k] > 0 ? "+" : ""}${importi[k]} ${MONETE.find((m) => m.key === k).short}`).join(", ");
  ChatMessage.create({
    content: `<p><b>💰 ${titolo} a ${righe.length} PG</b></p><ul>${righe.join("")}</ul>`
  });
  ui.notifications.info(`Monete assegnate a ${righe.length} PG.`);
})();
