rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // --- FUNZIONI DI SUPPORTO ---
    function isLoggedIn() {
      return request.auth != null;
    }

    function isMaster() {
      return isLoggedIn() && (
        request.auth.uid == "WNXsX7fSY5a3g2VZruYEXgFILOJ3" ||
        request.auth.token.email == "santomassimo85@gmail.com"
      );
    }

    // --- WORLD BOSS & TURNI ---
    match /battle_meta/turn_tracker {
      allow read: if isLoggedIn();
      allow create, delete: if isMaster();
      allow update: if isLoggedIn() && (
        isMaster() ||
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['actedPlayers', 'phase', 'expiryDate', 'turnNumber', 'attackCounts'])
      );
    }

    match /world_boss_system/{docId} {
      allow read: if isLoggedIn();
      allow write: if isMaster();
    }

    match /bosses/{bossId} {
      allow read: if true;
      allow create, delete: if isMaster();
      allow update: if isLoggedIn() && (
        isMaster() ||
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['hp', 'shield'])
      );
    }

    // --- PRESENZA ONLINE ---
    match /presence/{uid} {
      allow write: if request.auth != null && request.auth.uid == uid;
      allow read: if request.auth != null && request.auth.token.email == "santomassimo85@gmail.com";
    }

    // --- CHAT ---
    match /world_boss_chat/{msgId} {
      allow read: if isLoggedIn();
      allow create: if isLoggedIn();
      allow update, delete: if isMaster();
    }

    // Global chat: owner può cancellare i propri messaggi
    match /global_chat/{messageId} {
      allow read, create: if isLoggedIn();
      allow delete: if isLoggedIn() && (resource.data.uid == request.auth.uid || isMaster());
      allow update: if isMaster();
    }

    // --- PERSONAGGI & SCHEDE ---
    // I writes restano aperti perché il sistema Pet ha bisogno di scrivere
    // sul doc dell'avversario quando una battle viene risolta.
    match /characters/{charId} {
      allow read: if true;
      allow write: if true;
    }

    match /pg_sheets/{userId} {
      allow read, write: if isMaster() || (isLoggedIn() && request.auth.uid == userId);
    }

    // --- ARENA ---
    match /arena_config/{docId} {
      allow read: if isLoggedIn();
      allow write: if isMaster();
    }

    match /arena_meta/global {
      allow read: if isLoggedIn();
      allow write: if isLoggedIn();
    }

    match /characters_arena/{charId} {
      allow read: if isLoggedIn();
      allow write: if isLoggedIn() && (request.auth.uid + "_arena" == charId || isMaster());
    }

    match /arena_bets/{betId} {
      allow read: if isLoggedIn();
      allow create: if isLoggedIn();
      allow update: if isLoggedIn();
      allow delete: if isMaster();
    }

    match /arena_tournament_history/{docId} {
      allow read: if isMaster();
      allow create: if isLoggedIn();
      allow update, delete: if isMaster();
    }

    // --- PET ARENA (sistema compagni) ---
    // Sfide aperte e battaglie risolte tra giocatori. Lettura libera per
    // tutti gli utenti autenticati (leaderboard, log scontri); creazione
    // riservata allo sfidante; aggiornamento aperto perché chi accetta
    // scrive il risultato; cancellazione solo da parte dello sfidante.
    match /pet_battles/{battleId} {
      allow read: if isLoggedIn();
      allow create: if isLoggedIn() && request.resource.data.challenger.uid == request.auth.uid;
      allow update: if isLoggedIn();
      allow delete: if isLoggedIn() && (
        resource.data.challenger.uid == request.auth.uid ||
        isMaster()
      );
    }

    // --- FEEDBACK GIOCATORI ---
    // Una sola entry per (giocatore, funzionalità): doc id deterministico
    // `${uid}_${featureKey}` enforced via regola, così un utente non può
    // gonfiare la dashboard con copie multiple per la stessa feature.
    //   read   → Master legge tutto · player legge solo i propri (query
    //            client deve includere where("authorUid","==",uid))
    //   create/update → l'autore in request.resource.data deve coincidere
    //            con request.auth.uid, e il doc id deve iniziare con
    //            uid+"_". Le stelle restano vincolate a [0,5].
    //   delete → l'autore o il Master.
    match /feedback/{docId} {
      allow read: if isLoggedIn() && (
        isMaster() ||
        resource.data.authorUid == request.auth.uid
      );
      // Create: solo l'autore, sul proprio doc id deterministico,
      // con stelle nell'intervallo 0..5.
      allow create: if isLoggedIn()
        && request.resource.data.authorUid == request.auth.uid
        && docId == request.auth.uid + "_" + request.resource.data.feature
        && request.resource.data.stars is number
        && request.resource.data.stars >= 0
        && request.resource.data.stars <= 5;
      // Update: l'autore (con le stesse restrizioni) oppure il Master
      // (per scrivere masterReply / chiudere bug). Il Master non può
      // cambiare authorUid — la regola sotto impone che resti uguale.
      allow update: if isLoggedIn() && (
        (
          request.resource.data.authorUid == request.auth.uid
          && docId == request.auth.uid + "_" + request.resource.data.feature
          && request.resource.data.stars is number
          && request.resource.data.stars >= 0
          && request.resource.data.stars <= 5
        ) || (
          isMaster()
          && request.resource.data.authorUid == resource.data.authorUid
        )
      );
      allow delete: if isLoggedIn() && (
        resource.data.authorUid == request.auth.uid ||
        isMaster()
      );
    }

    // --- NOTIFICHE ---
    match /notifications/{msgId} {
      allow read: if isLoggedIn() && (resource.data.userId == request.auth.uid || isMaster());
      allow create: if isLoggedIn();
      allow update: if isLoggedIn() && (resource.data.userId == request.auth.uid || isMaster());
      allow delete: if isLoggedIn() && (resource.data.userId == request.auth.uid || isMaster());
    }

    // --- MERCATO & OGGETTI ---
    match /items/{itemId} {
      allow read: if true;
      allow create, delete: if isMaster();
      allow update: if isLoggedIn();
    }

    match /settings/market_config {
      allow read: if isLoggedIn();
      allow write: if isMaster();
    }

    // --- ALTRE COLLEZIONI ---
    match /geo_archive/{docId} { allow read: if true; allow write: if isMaster(); }
    match /map_pins/{pinId} { allow read: if true; allow write: if isMaster(); }
    match /npcs/{npcId} { allow read: if true; allow write: if isMaster(); }
    match /sessions/{partyId} { allow read: if true; allow write: if isMaster(); }
    match /quests/{questId} { allow read: if true; allow create, delete: if isMaster(); allow update: if isLoggedIn(); }
    match /session_videos/{video} { allow read: if isLoggedIn(); allow write: if isMaster(); }
    match /summaries/{summaryId} { allow read: if true; allow write: if isMaster(); allow update: if request.auth != null && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['viewCount']); }
    match /messages/{msgId} { allow create: if isLoggedIn(); allow read, write: if isMaster(); }
    match /party_tracks/{partyId} { allow read, write: if isLoggedIn(); }
    match /rolls/{rollId} { allow read, create: if isLoggedIn(); }
  }
}
